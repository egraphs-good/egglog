1//! Basic heuristics for whether or not to use a parallel or serial version of an algorithm.
2//!
3//! The parallel implementations in this crate generally have a noticeable overhead when compared
4//! to the serial versions on small problem sizes.
5
6use std::sync::OnceLock;
7
8const DEFAULT_DB_LEVEL_OP_CUTOFF: usize = 10_000;
9const DEFAULT_INDEX_CONSTRUCTION_CUTOFF: usize = 400_000;
10const DEFAULT_REBUILD_CUTOFF: usize = 400_000;
11const DEFAULT_INTRA_CONTAINER_CUTOFF: usize = 10_000;
12const DEFAULT_INTER_CONTAINER_CUTOFF: usize = 8;
13const DEFAULT_TABLE_OP_CUTOFF: usize = 400_000;
14const DEFAULT_FREE_JOIN_FORK_DEPTH: usize = 2;
15const DEFAULT_ACTION_BATCH_SIZE: usize = 8 * 1024;
16
17static CUTOFFS: OnceLock<Cutoffs> = OnceLock::new();
18
19struct Cutoffs {
20    db_level_op: usize,
21    index_construction: usize,
22    rebuild: usize,
23    intra_container: usize,
24    inter_container: usize,
25    table_op: usize,
26    free_join_fork_depth: usize,
27    action_batch_size: usize,
28}
29
30/// These are operations that work on a per-table or per-rule level where the size of the workload
31/// is hard to gauge ahead of time. In this case, we gate parallel execution based on the number of
32/// threads available and whether the total size of the database exceeds a certain threshold.
33pub(crate) fn parallelize_db_level_op(db_size: usize) -> bool {
34    should_parallelize(db_size, cutoffs().db_level_op)
35}
36
37/// Whether or not to use a parallel algorithm to construct a hash index.
38pub(crate) fn parallelize_index_construction(items_to_insert: usize) -> bool {
39    should_parallelize(items_to_insert, cutoffs().index_construction)
40}
41
42/// Whether or not to use a parallel algorithm to rebuild a [`crate::table::SortedWritesTable`].
43pub(crate) fn parallelize_rebuild(table_size: usize) -> bool {
44    should_parallelize(table_size, cutoffs().rebuild)
45}
46
47/// Whether or not to perform an operation for a given container memo table.
48pub(crate) fn parallelize_intra_container_op(num_containers: usize) -> bool {
49    should_parallelize(num_containers, cutoffs().intra_container)
50}
51
52/// Whether or not to perform an operation in parallel across a set of different container memo
53/// tables.
54pub(crate) fn parallelize_inter_container_op(num_containers: usize) -> bool {
55    should_parallelize(num_containers, cutoffs().inter_container)
56}
57
58#[track_caller]
59pub(crate) fn parallelize_table_op(table_size: usize) -> bool {
60    should_parallelize(table_size, cutoffs().table_op)
61}
62
63/// Number of top free-join frames that may fork recursive drain work.
64pub(crate) fn free_join_fork_depth() -> usize {
65    cutoffs().free_join_fork_depth
66}
67
68/// Number of action bindings to batch before dispatching a scoped worker task.
69pub(crate) fn action_batch_size() -> usize {
70    cutoffs().action_batch_size
71}
72
73fn should_parallelize(len: usize, cutoff: usize) -> bool {
74    len > cutoff && crate::parallel::current_num_threads() > 1
75}
76
77fn cutoffs() -> &'static Cutoffs {
78    CUTOFFS.get_or_init(|| Cutoffs {
79        db_level_op: cutoff(
80            "EGGLOG_PARALLEL_DB_LEVEL_OP_CUTOFF",
81            DEFAULT_DB_LEVEL_OP_CUTOFF,
82        ),
83        index_construction: cutoff(
84            "EGGLOG_PARALLEL_INDEX_CONSTRUCTION_CUTOFF",
85            DEFAULT_INDEX_CONSTRUCTION_CUTOFF,
86        ),
87        rebuild: cutoff("EGGLOG_PARALLEL_REBUILD_CUTOFF", DEFAULT_REBUILD_CUTOFF),
88        intra_container: cutoff(
89            "EGGLOG_PARALLEL_INTRA_CONTAINER_CUTOFF",
90            DEFAULT_INTRA_CONTAINER_CUTOFF,
91        ),
92        inter_container: cutoff(
93            "EGGLOG_PARALLEL_INTER_CONTAINER_CUTOFF",
94            DEFAULT_INTER_CONTAINER_CUTOFF,
95        ),
96        table_op: cutoff("EGGLOG_PARALLEL_TABLE_OP_CUTOFF", DEFAULT_TABLE_OP_CUTOFF),
97        free_join_fork_depth: cutoff(
98            "EGGLOG_PARALLEL_FREE_JOIN_FORK_DEPTH",
99            DEFAULT_FREE_JOIN_FORK_DEPTH,
100        ),
101        action_batch_size: cutoff(
102            "EGGLOG_PARALLEL_ACTION_BATCH_SIZE",
103            DEFAULT_ACTION_BATCH_SIZE,
104        )
105        .max(1),
106    })
107}
108
109fn cutoff(name: &str, default: usize) -> usize {
110    std::env::var(name)
111        .ok()
112        .and_then(|value| value.parse().ok())
113        .unwrap_or(default)
114}
