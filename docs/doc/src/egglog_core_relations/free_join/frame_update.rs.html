1//! A data-structure for low-overhead buffering of updates in a free join
2//! execution.
3//!
4//! Free Join is a recursive algorithm that that discovers a candidate binding for a particular
5//! variable in a query and then recursively runs the rest of the join restricted for that binding
6//! holding. Once the "sub-join" finishes, the outer recursive call backtracks, adds a separate
7//! binding, and then repeats.
8//!
9//! The Free Join paper observed that this resulted in poor cache behavior because for every cell
10//! iterated over in an outer stage, we had to do several other steps on successive inner stages.
11//! Instead, we can accumulate a set of new bindings in a separate buffer and then iterate over
12//! those bindings in recursive calls. When parallelism is enabled, this data-structure allows us
13//! hand over an entire batch of recursive calls to a separate thread to process independently.
14
15use crate::free_join::probe::AtomRows;
16use crate::offsets::OffsetRange;
17
18use crate::Value;
19
20use super::{AtomId, Variable};
21
22#[derive(Debug)]
23pub(super) enum UpdateInstr<'rows, 'exec> {
24    PushBinding(Variable, Value),
25    RefineAtom(AtomId, AtomRows<'rows, 'exec>),
26    /// Refine an atom to a dense offset range, avoiding an Arc<TrieRoot> allocation.
27    RefineAtomDense(AtomId, OffsetRange),
28    /// Marks the end of the current frame. Time to make a recursive call.
29    EndFrame,
30}
31
32/// Compact instruction stored in [`FrameUpdates`]. Variable-sized atom rows
33/// live in a side buffer so the common binding and frame-marker instructions
34/// do not inherit the size of [`AtomRows`]'s largest variant.
35#[derive(Debug)]
36enum BufferedUpdateInstr {
37    PushBinding(Variable, Value),
38    RefineAtom(AtomId),
39    RefineAtomDense(AtomId, OffsetRange),
40    EndFrame,
41}
42
43/// A flat buffer of updates that is used to prepare a sequence of recursive calls to free join.
44#[derive(Default)]
45pub(super) struct FrameUpdates<'rows, 'exec> {
46    updates: Vec<BufferedUpdateInstr>,
47    subsets: Vec<AtomRows<'rows, 'exec>>,
48    frames: usize,
49    last_start: usize,
50    last_subset_start: usize,
51}
52
53impl<'rows, 'exec> FrameUpdates<'rows, 'exec> {
54    pub(super) fn with_capacity(capacity: usize) -> FrameUpdates<'rows, 'exec> {
55        FrameUpdates {
56            // A two-prober frame commonly contains one binding, two atom
57            // refinements, and the end marker.
58            updates: Vec::with_capacity(capacity.saturating_mul(4)),
59            subsets: Vec::with_capacity(capacity.saturating_mul(2)),
60            frames: 0,
61            last_start: 0,
62            last_subset_start: 0,
63        }
64    }
65
66    /// Bind `var` to `val` in the current frame.
67    pub(super) fn push_binding(&mut self, var: Variable, val: Value) {
68        self.updates
69            .push(BufferedUpdateInstr::PushBinding(var, val));
70    }
71
72    /// Refine `atom` to consider only the given `subset` in the current frame.
73    pub(super) fn refine_atom(&mut self, atom: AtomId, node: impl Into<AtomRows<'rows, 'exec>>) {
74        self.subsets.push(node.into());
75        self.updates.push(BufferedUpdateInstr::RefineAtom(atom));
76    }
77
78    /// Refine `atom` to consider only the given dense offset range, without
79    /// allocating an Arc<TrieRoot> eagerly.
80    pub(super) fn refine_atom_dense(&mut self, atom: AtomId, range: OffsetRange) {
81        self.updates
82            .push(BufferedUpdateInstr::RefineAtomDense(atom, range));
83    }
84
85    /// Roll back the updates to the last frame start. Note that repeated calls
86    /// to this method will still only roll back one frame (total).
87    pub(super) fn rollback(&mut self) {
88        self.updates.truncate(self.last_start);
89        self.subsets.truncate(self.last_subset_start);
90    }
91
92    /// Finish the current frame and prepare for the next one.
93    pub(super) fn finish_frame(&mut self) {
94        self.updates.push(BufferedUpdateInstr::EndFrame);
95        self.last_start = self.updates.len();
96        self.last_subset_start = self.subsets.len();
97        self.frames += 1;
98    }
99
100    /// Get the number of frames that have been finished.
101    pub(super) fn frames(&self) -> usize {
102        self.frames
103    }
104
105    pub(super) fn clear(&mut self) {
106        self.updates.clear();
107        self.subsets.clear();
108        self.frames = 0;
109        self.last_start = 0;
110        self.last_subset_start = 0;
111    }
112
113    pub(super) fn drain(&mut self, mut f: impl FnMut(UpdateInstr<'rows, 'exec>)) {
114        // Reset bookkeeping before invoking user code. Both drains still own
115        // and drop their remaining elements if `f` unwinds, leaving the
116        // buffer reusable after a caught panic.
117        self.frames = 0;
118        self.last_start = 0;
119        self.last_subset_start = 0;
120        {
121            let mut subsets = self.subsets.drain(..);
122            for update in self.updates.drain(..) {
123                let update = match update {
124                    BufferedUpdateInstr::PushBinding(var, val) => {
125                        UpdateInstr::PushBinding(var, val)
126                    }
127                    BufferedUpdateInstr::RefineAtom(atom) => {
128                        let rows = subsets
129                            .next()
130                            .expect("refine instruction is missing its atom rows");
131                        UpdateInstr::RefineAtom(atom, rows)
132                    }
133                    BufferedUpdateInstr::RefineAtomDense(atom, range) => {
134                        UpdateInstr::RefineAtomDense(atom, range)
135                    }
136                    BufferedUpdateInstr::EndFrame => UpdateInstr::EndFrame,
137                };
138                f(update);
139            }
140            debug_assert!(subsets.next().is_none());
141        }
142    }
143}
