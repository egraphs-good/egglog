1//! Proof checking for egglog proofs.
2//! Given an egglog program and a proof for some proposition, check that the proof is valid.
3//! The main work is to check proofs for rules, ensuring that under the substitution
4//! the rule would have matched the propositions for each premise, and that the conclusion
5//! is contained in the rule's actions.
6//! For checking queries and actions, we evaluate the expressions under a substitution
7//! and produce a set of valid propositions with `process_actions`.
8//! The set of valid propositions includes reflexive equalities for all subterms.
9
10use crate::{
11    Term, TermDag, TermId,
12    ast::{
13        FunctionSubtype, GenericAction, GenericNCommand, ResolvedExpr, ResolvedFact,
14        ResolvedNCommand,
15    },
16    core::ResolvedCall,
17    proofs::proof_format::{Justification, ProofId, ProofStore, Proposition},
18    util::{HashMap, HashSet, SymbolGen},
19};
20use thiserror::Error;

22/// A side condition is a rule-body fact that is just a container-producing
23/// primitive applied to bound variables — `(= v (vec-of e))`, `(= (set-of a)
24/// (set-of b))`, etc. Its proof is a bare [`Justification::Eval`] marker and it
25/// is verified by re-evaluation in [`ProofStore::check_side_condition`] rather
26/// than by matching a premise proposition.
27pub(super) fn is_container_side_condition(fact: &ResolvedFact) -> bool {
28    fn is_container_primitive(expr: &ResolvedExpr) -> bool {
29        matches!(
30            expr,
31            ResolvedExpr::Call(_, ResolvedCall::Primitive(p), _)
32                if p.output().is_eq_container_sort()
33        )
34    }
35    match fact {
36        ResolvedFact::Eq(_, lhs, rhs) => is_container_primitive(lhs) || is_container_primitive(rhs),
37        ResolvedFact::Fact(expr) => is_container_primitive(expr),
38    }
39}

50/// Gathers all global CoreActions from a program.
51/// This extracts all actions that occur at the top level, filtering out NormRule and other commands.
52pub(crate) fn gather_global_actions(
53    prog: &[ResolvedNCommand],
54) -> impl Iterator<Item = &GenericAction<ResolvedCall, crate::ast::ResolvedVar>> {
55    prog.iter().filter_map(|cmd| {
56        if let GenericNCommand::CoreAction(action) = cmd {
57            Some(action)
58        } else {
59            None
60        }
61    })
62}

64/// Run a merge function and return the resulting term, as well as a set of propositions learned.
65pub(crate) fn run_merge(
66    term_dag: &mut TermDag,
67    func_name: &str,
68    prog: &[ResolvedNCommand],
69    old_term: TermId,
70    new_term: TermId,
71) -> Result<(TermId, HashSet<Proposition>), ProofCheckError> {
72    let mut subst = HashMap::default();
73    subst.insert("old".to_string(), old_term);
74    subst.insert("new".to_string(), new_term);
75    for cmd in prog {
76        if let GenericNCommand::Function(func_decl) = cmd
77            && func_decl.name == func_name
78        {
79            // run the merge function for this function using eval_expr
80            let expr = func_decl.merge.as_ref().ok_or_else(|| {
81                ProofCheckError::from(ProofCheckErrorKind::FunctionNotFound {
82                    function_name: func_name.to_string(),
83                })
84            })?;
85            return eval_expr_with_subst("merge_function", expr, term_dag, &subst);
86        }
87    }
88    Err(ProofCheckErrorKind::FunctionNotFound {
89        function_name: func_name.to_string(),
90    }
91    .into())
92}

94/// Given a sequence of actions, computes:
95/// 1. All the terms bound to variables (from Let actions)
96/// 2. All the propositions implied by the actions:
97///    - Reflexive equalities for all subterms
98///    - Ground equalities from union statements (bidirectional)
99///    - Reflexive equalities from set statements
100pub(crate) fn process_actions(
101    rule_name: &str,
102    mut bindings: HashMap<String, TermId>,
103    actions: &[&GenericAction<ResolvedCall, crate::ast::ResolvedVar>],
104    term_dag: &mut TermDag,
105) -> Result<ActionContext, ProofCheckError> {
106    let mut propositions = HashSet::default();
107
108    // Single pass: process all actions, accumulating bindings and propositions
109    for action in actions {
110        match action {
111            GenericAction::Let(_, var, expr) => {
112                // Evaluate the expression and collect propositions
113                let (term_id, new_props) =
114                    eval_expr_with_subst(rule_name, expr, term_dag, &bindings)?;
115                bindings.insert(var.name.clone(), term_id);
116                propositions.extend(new_props);
117            }
118            GenericAction::Union(_, lhs_expr, rhs_expr) => {
119                // Union creates ground equalities
120                let (lhs_term, lhs_props) =
121                    eval_expr_with_subst(rule_name, lhs_expr, term_dag, &bindings)?;
122                let (rhs_term, rhs_props) =
123                    eval_expr_with_subst(rule_name, rhs_expr, term_dag, &bindings)?;
124
125                // Collect propositions from evaluating both sides
126                propositions.extend(lhs_props);
127                propositions.extend(rhs_props);
128                // Store both directions of the equality
129                propositions.insert(Proposition::new(lhs_term, rhs_term));
130                propositions.insert(Proposition::new(rhs_term, lhs_term));
131            }
132            GenericAction::Set(_, func, args, rhs) => {
133                // Set creates reflexive equality for the resulting term
134                let mut all_args = args.to_vec();
135                all_args.push(rhs.clone());
136                let call_expr = ResolvedExpr::Call(crate::ast::Span::Panic, func.clone(), all_args);
137                let (_term, new_props) =
138                    eval_expr_with_subst(rule_name, &call_expr, term_dag, &bindings)?;
139                propositions.extend(new_props);
140            }
141            GenericAction::Expr(_, expr) => {
142                // Expr creates reflexive equality for its result
143                let (_, new_props) = eval_expr_with_subst(rule_name, expr, term_dag, &bindings)?;
144                propositions.extend(new_props);
145            }
146            GenericAction::Panic(_, _) => {
147                // Panics do not create propositions
148            }
149            GenericAction::Change(_, _, _, _) => {
150                // Changes do not create propositions
151            }
152        }
153    }
154
155    Ok(ActionContext {
156        var_bindings: bindings,
157        propositions,
158    })
159}

161/// Evaluate an expression under a substitution.
162/// Returns Ok((TermId, propositions)) if successful, where propositions include
163/// all reflexive equalities for the term and its subterms.
164/// Returns Err(()) if evaluation fails.
165fn eval_expr_with_subst(
166    rule_name: &str,
167    expr: &ResolvedExpr,
168    dag: &mut TermDag,
169    subst: &HashMap<String, TermId>,
170) -> Result<(TermId, HashSet<Proposition>), ProofCheckError> {
171    let mut propositions = HashSet::default();
172
173    let term_id = match expr {
174        ResolvedExpr::Lit(_, lit) => dag.lit(lit.clone()),
175        ResolvedExpr::Var(_, var) => subst.get(&var.name).copied().ok_or_else(|| {
176            ProofCheckError::from(ProofCheckErrorKind::UnboundVariable {
177                rule_name: rule_name.to_string(),
178                variable: var.name.clone(),
179                available: subst.keys().cloned().collect::<Vec<_>>().join(", "),
180            })
181        })?,
182        ResolvedExpr::Call(_, head, args) => match head {
183            ResolvedCall::Func(_func_type) => {
184                let mut arg_terms = Vec::new();
185                for arg in args {
186                    let (arg_term, arg_props) = eval_expr_with_subst(rule_name, arg, dag, subst)?;
187                    arg_terms.push(arg_term);
188                    propositions.extend(arg_props);
189                }
190                dag.app(head.name().to_string(), arg_terms)
191            }
192            ResolvedCall::Primitive(specialized_primitive) => {
193                // run validator, throwing error if it fails
194                let mut arg_terms = Vec::new();
195                for arg in args {
196                    let (arg_term, arg_props) = eval_expr_with_subst(rule_name, arg, dag, subst)?;
197                    arg_terms.push(arg_term);
198                    propositions.extend(arg_props);
199                }
200                // checked by earlier code showing this program supports proofs
201                let validator = specialized_primitive
202                    .validator()
203                    .expect("Expected primitive to have validator since proof mode is enabled");
204                validator(dag, &arg_terms).ok_or_else(|| {
205                    ProofCheckError::from(ProofCheckErrorKind::PrimitiveValidatorFailed {
206                        function_name: specialized_primitive.name().to_string(),
207                    })
208                })?
209            }
210        },
211    };
212
213    // Add reflexive equality for this term and all its subterms
214    add_subterm_reflexive_equalities(term_id, dag, &mut propositions);
215
216    Ok((term_id, propositions))
217}

219/// Add reflexive equalities for all subterms of a term.
220/// For example, if we have proved `(f (g 1)) = (f (g 1))`, then we have also proved
221// that `(g 1) = (g 1)` and `1 = 1`.
222fn add_subterm_reflexive_equalities(
223    term_id: TermId,
224    term_dag: &TermDag,
225    propositions: &mut HashSet<Proposition>,
226) {
227    add_subterm_reflexive_equalities_helper(
228        term_id,
229        term_dag,
230        propositions,
231        &mut Default::default(),
232    );
233}
234
235fn add_subterm_reflexive_equalities_helper(
236    term_id: TermId,
237    term_dag: &TermDag,
238    propositions: &mut HashSet<Proposition>,
239    seen: &mut HashSet<TermId>,
240) {
241    if !seen.insert(term_id) {
242        return;
243    }
244
245    // Add reflexive equality for this term
246    propositions.insert(Proposition::new(term_id, term_id));
247
248    // Recursively add for all children
249    if let Term::App(_, children) = term_dag.get(term_id) {
250        for &child_id in children {
251            add_subterm_reflexive_equalities_helper(child_id, term_dag, propositions, seen);
252        }
253    }
254}

256/// Gathers all global variables from a program and computes their values as terms
257/// without using globals (i.e., all global references are replaced with their definitions).
258///
259/// This expects the program to be in proof normalized form where globals appear as Let actions.
260pub(crate) fn gather_globals(
261    prog: &[ResolvedNCommand],
262    term_dag: &mut TermDag,
263) -> Result<HashMap<String, TermId>, ProofCheckError> {
264    let actions: Vec<_> = gather_global_actions(prog).collect();
265    let ctx = process_actions("global_action", HashMap::default(), &actions, term_dag)?;
266    Ok(ctx.var_bindings)
267}

288/// The kinds of errors that can occur during proof checking
289#[derive(Debug, Clone, Error)]
290pub enum ProofCheckErrorKind {
291    /// The proof claims terms are equal but they don't match the actual terms
292    #[error(
293        "Proof {proof_id} claims to prove {expected_lhs:?} = {expected_rhs:?}, but actually proves {actual_lhs:?} = {actual_rhs:?}"
294    )]
295    TermMismatch {
296        proof_id: ProofId,
297        expected_lhs: TermId,
298        expected_rhs: TermId,
299        actual_lhs: TermId,
300        actual_rhs: TermId,
301    },
302    /// Transitivity requires matching middle terms
303    #[error(
304        "Proof {proof_id}: transitivity requires matching middle terms, but left.rhs = {left_rhs:?} and right.lhs = {right_lhs:?}"
305    )]
306    TransitivityMismatch {
307        proof_id: ProofId,
308        left_rhs: TermId,
309        right_lhs: TermId,
310    },
311    /// Congruence proof: base rhs is not a function application
312    #[error("Proof {proof_id}: congruence error - base proof rhs is not a function application")]
313    CongruenceBaseNotApp { proof_id: ProofId },
314    /// Congruence proof: child index out of bounds
315    #[error(
316        "Proof {proof_id}: congruence error - child index {child_index} out of bounds for term with {num_children} children"
317    )]
318    CongruenceChildIndexOutOfBounds {
319        proof_id: ProofId,
320        child_index: usize,
321        num_children: usize,
322    },
323    /// Congruence proof: child proof lhs doesn't match base term child
324    #[error(
325        "Proof {proof_id}: congruence error - child proof lhs {child_lhs:?} doesn't match base term child {base_child:?} at index {child_index}"
326    )]
327    CongruenceChildMismatch {
328        proof_id: ProofId,
329        child_lhs: TermId,
330        base_child: TermId,
331        child_index: usize,
332    },
333    /// Congruence proof: result doesn't match expected
334    #[error(
335        "Proof {proof_id}: congruence error - proof rhs {proof_rhs:?} doesn't match expected {expected_rhs:?}"
336    )]
337    CongruenceResultMismatch {
338        proof_id: ProofId,
339        proof_rhs: TermId,
340        expected_rhs: TermId,
341    },
342    /// Congruence proof: lhs doesn't match base proof lhs
343    #[error("Proof {proof_id}: congruence error - proof lhs doesn't match base proof lhs")]
344    CongruenceLhsMismatch { proof_id: ProofId },
345    /// Rule application has wrong number of premises
346    #[error(
347        "Proof {proof_id}: rule '{rule_name}' expects {expected} premises, but proof has {actual}"
348    )]
349    RulePremiseCountMismatch {
350        proof_id: ProofId,
351        rule_name: String,
352        expected: usize,
353        actual: usize,
354    },
355    /// Variable not found in substitution during proof checking
356    #[error(
357        "Rule '{rule_name}': variable '{variable}' not found in substitution. Available: {available}"
358    )]
359    UnboundVariable {
360        rule_name: String,
361        variable: String,
362        available: String,
363    },
364    /// Function fact doesn't match the expected reflexive equality proposition
365    #[error(
366        "Rule '{rule_name}': function fact mismatch - expected reflexive equality for {expected}, got {actual_lhs} = {actual_rhs}"
367    )]
368    FunctionFactMismatch {
369        rule_name: String,
370        expected: String,
371        actual_lhs: String,
372        actual_rhs: String,
373    },
374    /// Equality fact doesn't match the proven proposition under substitution
375    #[error(
376        "Rule '{rule_name}': equality fact mismatch under substitution.\nFact: {fact}\nSubstituted: (= {substituted_lhs} {substituted_rhs})\nPremise proves: (= {proven_lhs} {proven_rhs})"
377    )]
378    EqualityFactMismatch {
379        rule_name: String,
380        fact: String,
381        substituted_lhs: String,
382        substituted_rhs: String,
383        proven_lhs: String,
384        proven_rhs: String,
385    },
386    /// Plain fact expression doesn't match proposition under substitution
387    #[error(
388        "Rule '{rule_name}': fact mismatch - {fact} under substitution {substitution} gives {actual}, expected {expected}"
389    )]
390    FactMismatch {
391        rule_name: String,
392        fact: String,
393        substitution: String,
394        actual: String,
395        expected: String,
396    },
397    /// Rule head actions don't produce the claimed equality
398    #[error(
399        "Rule '{rule_name}': rule head doesn't produce claimed equality.\nLHS: {claimed_lhs}\nRHS: {claimed_rhs}\nSubstitution: {substitution}"
400    )]
401    RuleHeadMismatch {
402        rule_name: String,
403        claimed_lhs: String,
404        claimed_rhs: String,
405        substitution: String,
406    },
407    /// Primitive operation validator failed
408    #[error("Primitive '{function_name}' validation failed")]
409    PrimitiveValidatorFailed { function_name: String },
410    /// Primitive has no validator for proof checking

461        claimed_rhs: String,

513/// Context needed for proof checking
514pub(crate) struct ProofCheckContext {
515    /// Set of equalities established by global union/set actions
516    /// Each entry is a pair (lhs, rhs) that was unified
517    /// This includes reflexive equalities (term, term) for all globals
518    global_equalities: HashSet<Proposition>,
519    /// Map of global variable names to their TermIds
520    global_bindings: HashMap<String, TermId>,
521    /// Cache of already-checked proofs
522    checked_proofs: HashMap<ProofId, Proposition>,
523}
524
525impl ProofCheckContext {
526    /// Create a new proof check context by analyzing the program.
527    /// This gathers all equalities established by global actions (unions and sets).
528    fn new(prog: &[ResolvedNCommand], term_dag: &mut TermDag) -> Result<Self, ProofCheckError> {
529        // Check for duplicate rule names
530        let mut seen_rule_names: HashSet<&str> = HashSet::default();
531        for cmd in prog {
532            if let GenericNCommand::NormRule { rule } = cmd
533                && !seen_rule_names.insert(&rule.name)
534            {
535                return Err(ProofCheckErrorKind::DuplicateRuleName {
536                    rule_name: rule.name.clone(),
537                }
538                .into());
539            }
540        }
541
542        // Use the new refactored functions
543        let actions: Vec<_> = gather_global_actions(prog).collect();
544        let action_ctx = process_actions("global_actions", HashMap::default(), &actions, term_dag)?;
545
546        Ok(ProofCheckContext {
547            global_equalities: action_ctx.propositions,
548            checked_proofs: HashMap::default(),
549            global_bindings: action_ctx.var_bindings,
550        })
551    }
552
553    fn in_globals(&self, lhs: TermId, rhs: TermId) -> bool {
554        self.global_equalities.contains(&Proposition::new(lhs, rhs))
555    }
556}

558/// Helper function to format a term with let bindings
559fn format_term(term_dag: &TermDag, term_id: TermId) -> String {
560    term_dag.to_string_with_let(&mut SymbolGen::new("".to_string()), term_id)

563/// Helper function to format a substitution as a string
564fn format_substitution(term_dag: &TermDag, substitution: &HashMap<String, TermId>) -> String {
565    substitution
566        .iter()
567        .map(|(k, v)| format!("{} -> {}", k, format_term(term_dag, *v)))
568        .collect::<Vec<_>>()
569        .join(", ")
570}

572impl ProofStore {
573    /// Check that a proof is valid with respect to a typechecked program.
574    pub(crate) fn check_proof(
575        &mut self,
576        proof_id: ProofId,
577        program: &[ResolvedNCommand],
578    ) -> Result<Proposition, ProofCheckError> {
579        let mut ctx = ProofCheckContext::new(program, &mut self.term_dag)?;
580        self.check_proof_with_context(proof_id, program, &mut ctx)
581    }
582
583    /// Internal recursive proof checker with context
584    fn check_proof_with_context(
585        &mut self,
586        proof_id: ProofId,
587        program: &[ResolvedNCommand],
588        ctx: &mut ProofCheckContext,
589    ) -> Result<Proposition, ProofCheckError> {
590        // Check cache first
591        if let Some(prop) = ctx.checked_proofs.get(&proof_id) {
592            return Ok(prop.clone());
593        }
594
595        let proof = self.id_to_proof[proof_id].clone();
596        let result = match &proof.justification {
597            Justification::Fiat => {
598                // if the both terms are primitives and equal, accept
599                let term = self.term_dag.get(proof.lhs());
600                if (matches!(term, Term::Lit(_)) && proof.lhs() == proof.rhs())
601                    || ctx.in_globals(proof.lhs(), proof.rhs())
602                {
603                    Ok(Proposition::new(proof.lhs(), proof.rhs()))
604                } else {
605                    Err(ProofCheckErrorKind::InvalidFiat {
606                        proof_id,
607                        lhs: proof.lhs(),
608                        rhs: proof.rhs(),
609                    }
610                    .into())
611                }
612            }
613
614            Justification::Rule {
615                name,
616                premise_proofs,
617                substitution,
618            } => {
619                // Find the rule in the program
620                let rule = program
621                    .iter()
622                    .find_map(|cmd| match cmd {
623                        GenericNCommand::NormRule { rule } if &rule.name == name => Some(rule),
624                        _ => None,
625                    })
626                    .ok_or_else(|| {
627                        ProofCheckError::from(ProofCheckErrorKind::RuleNotFound {
628                            rule_name: name.clone(),
629                        })
630                    })?;
631
632                // Check premise count
633                if rule.body.len() != premise_proofs.len() {
634                    return Err(ProofCheckErrorKind::RulePremiseCountMismatch {
635                        proof_id,
636                        rule_name: name.clone(),
637                        expected: rule.body.len(),
638                        actual: premise_proofs.len(),
639                    }
640                    .into());
641                }
642
643                let mut working_subst = ctx
644                    .global_bindings
645                    .iter()
646                    .map(|(k, v)| (k.clone(), *v))
647                    .chain(substitution.iter().map(|(k, v)| (k.clone(), *v)))
648                    .collect::<HashMap<_, _>>();
649
650                // Verify each premise in order. A container side condition carries
651                // only an `Eval` marker, so re-evaluate it here with the rule's
652                // typed validator — this binds its output into the substitution for
653                // later facts. Every other fact is matched against its premise
654                // proposition.
655                for (fact, &premise_id) in rule.body.iter().zip(premise_proofs.iter()) {
656                    self.assert_body_proof_normal_form(fact, name)?;
657                    if is_container_side_condition(fact) {
658                        self.check_side_condition(fact, &mut working_subst, name)?;
659                    } else {
660                        let prop = self.check_proof_with_context(premise_id, program, ctx)?;
661                        self.check_fact_matches_proposition(fact, &prop, &working_subst, name)?;
662                    }
663                }
664
665                // Verify that the conclusion matches what the rule produces
666                self.check_rule_produces_equality(
667                    rule,
668                    substitution,
669                    &working_subst,
670                    proof.proposition(),
671                    name,
672                )?;
673
674                Ok(Proposition::new(proof.lhs(), proof.rhs()))
675            }
676
677            Justification::MergeFn {
678                function,
679                old_proof,
680                new_proof,
681            } => {
682                // Check both sub-proofs - they should be reflexive proofs
683                let old_prop = self.check_proof_with_context(*old_proof, program, ctx)?;
684                let new_prop = self.check_proof_with_context(*new_proof, program, ctx)?;
685
686                let (old_lhs, old_rhs) = (old_prop.lhs, old_prop.rhs);
687                let (new_lhs, new_rhs) = (new_prop.lhs, new_prop.rhs);
688
689                // MergeFn proofs expect reflexive equality proofs
690                if old_lhs != old_rhs {
691                    return Err(ProofCheckErrorKind::MergeFnNotReflexive {
692                        proof_id,
693                        which: "old".to_string(),
694                        lhs: old_lhs,
695                        rhs: old_rhs,
696                    }
697                    .into());
698                }
699                if new_lhs != new_rhs {
700                    return Err(ProofCheckErrorKind::MergeFnNotReflexive {
701                        proof_id,
702                        which: "new".to_string(),
703                        lhs: new_lhs,
704                        rhs: new_rhs,
705                    }
706                    .into());
707                }
708
709                let old_view_term = self.term_dag.get(old_rhs);
710                let new_view_term = self.term_dag.get(new_rhs);
711
712                let (old_term, new_term, view_head, input_args) =
713                    match (old_view_term.clone(), new_view_term.clone()) {
714                        (Term::App(old_head, old_args), Term::App(new_head, new_args)) => {
715                            // Verify both are views of the same function
716                            if old_head != new_head {
717                                return Err(ProofCheckErrorKind::MergeFnFunctionMismatch {
718                                    proof_id,
719                                    old_func: old_head.clone(),
720                                    new_func: new_head.clone(),
721                                }
722                                .into());
723                            }
724                            // The last argument is the output
725                            let old_output = *old_args.last().ok_or_else(|| {
726                                ProofCheckError::from(ProofCheckErrorKind::MergeFnEmptyArgs {
727                                    proof_id,
728                                    which: "old".to_string(),
729                                })
730                            })?;
731                            let new_output = *new_args.last().ok_or_else(|| {
732                                ProofCheckError::from(ProofCheckErrorKind::MergeFnEmptyArgs {
733                                    proof_id,
734                                    which: "new".to_string(),
735                                })
736                            })?;
737                            // Get the input arguments (all but the last)
738                            let inputs: Vec<TermId> = old_args[..old_args.len() - 1].to_vec();
739                            // inputs should match for old and new
740                            if inputs.len() != new_args.len() - 1
741                                || inputs
742                                    .iter()
743                                    .zip(new_args[..new_args.len() - 1].iter())
744                                    .any(|(a, b)| a != b)
745                            {
746                                return Err(
747                                    ProofCheckErrorKind::MergeFnInputMismatch { proof_id }.into()
748                                );
749                            }
750
751                            (old_output, new_output, old_head.clone(), inputs)
752                        }
753                        _ => {
754                            return Err(ProofCheckErrorKind::MergeFnNotApp {
755                                proof_id,
756                                old_term: old_rhs,
757                                new_term: new_rhs,
758                            }
759                            .into());
760                        }
761                    };
762
763                // Run the merge function to get the expected result
764                let (merged_term_child, mut merged_props) =
765                    run_merge(&mut self.term_dag, function, program, old_term, new_term)?;
766                // Add f(inputs..., merged_term) to merged_props
767                let mut merged_view_args = input_args.clone();
768                merged_view_args.push(merged_term_child);
769                let merged_term = self.term_dag.app(view_head, merged_view_args);
770                merged_props.insert(Proposition::new(merged_term, merged_term));
771                // Verify the proof's claimed equality is in the merged propositions
772                if !merged_props.contains(&Proposition::new(proof.lhs(), proof.rhs())) {
773                    return Err(ProofCheckErrorKind::MergeFnResultMismatch {
774                        proof_id,
775                        claimed_lhs: self
776                            .term_dag
777                            .to_string_with_let(&mut SymbolGen::new("".to_string()), proof.lhs()),
778                        claimed_rhs: self
779                            .term_dag
780                            .to_string_with_let(&mut SymbolGen::new("".to_string()), proof.rhs()),
781                    }
782                    .into());
783                }
784
785                Ok(Proposition::new(proof.lhs(), proof.rhs()))
786            }
787
788            Justification::Trans(left_id, right_id) => {
789                // Check both sub-proofs
790                let left_prop = self.check_proof_with_context(*left_id, program, ctx)?;
791                let right_prop = self.check_proof_with_context(*right_id, program, ctx)?;
792
793                let (left_lhs, left_rhs) = (left_prop.lhs, left_prop.rhs);
794                let (right_lhs, right_rhs) = (right_prop.lhs, right_prop.rhs);
795
796                // Check transitivity: left.rhs must equal right.lhs
797                if left_rhs != right_lhs {
798                    return Err(ProofCheckErrorKind::TransitivityMismatch {
799                        proof_id,
800                        left_rhs,
801                        right_lhs,
802                    }
803                    .into());
804                }
805
806                // Result should be left_lhs = right_rhs
807                if proof.lhs() != left_lhs || proof.rhs() != right_rhs {
808                    return Err(ProofCheckErrorKind::TermMismatch {
809                        proof_id,
810                        expected_lhs: proof.lhs(),
811                        expected_rhs: proof.rhs(),
812                        actual_lhs: left_lhs,
813                        actual_rhs: right_rhs,
814                    }
815                    .into());
816                }
817
818                Ok(Proposition::new(proof.lhs(), proof.rhs()))
819            }
820
821            Justification::Sym(inner_id) => {
822                // Check the inner proof
823                let inner_prop = self.check_proof_with_context(*inner_id, program, ctx)?;
824                let (inner_lhs, inner_rhs) = (inner_prop.lhs, inner_prop.rhs);
825
826                // Symmetry swaps lhs and rhs
827                if proof.lhs() != inner_rhs || proof.rhs() != inner_lhs {
828                    return Err(ProofCheckErrorKind::TermMismatch {
829                        proof_id,
830                        expected_lhs: proof.lhs(),
831                        expected_rhs: proof.rhs(),
832                        actual_lhs: inner_rhs,
833                        actual_rhs: inner_lhs,
834                    }
835                    .into());
836                }
837
838                Ok(Proposition::new(proof.lhs(), proof.rhs()))
839            }
840
841            Justification::Congr {
842                proof: base_id,
843                child_index,
844                child_proof: child_id,
845            } => {
846                // Check the base proof (proves t1 = f(..., ci, ...))
847                let base_prop = self.check_proof_with_context(*base_id, program, ctx)?;
848                let (base_lhs, base_rhs) = (base_prop.lhs, base_prop.rhs);
849
850                // Check the child proof (proves ci = c2)
851                let child_prop = self.check_proof_with_context(*child_id, program, ctx)?;
852                let (child_lhs, child_rhs) = (child_prop.lhs, child_prop.rhs);
853
854                // base_rhs should be an application f(...)
855                let (func_name, children) = match self.term_dag.get(base_rhs) {
856                    Term::App(f, cs) => (f.clone(), cs.clone()),
857                    _ => {
858                        return Err(ProofCheckErrorKind::CongruenceBaseNotApp { proof_id }.into());
859                    }
860                };
861
862                // Check child_index is valid
863                if *child_index >= children.len() {
864                    return Err(ProofCheckErrorKind::CongruenceChildIndexOutOfBounds {
865                        proof_id,
866                        child_index: *child_index,
867                        num_children: children.len(),
868                    }
869                    .into());
870                }
871
872                // Check that child_lhs matches the child at child_index
873                if children[*child_index] != child_lhs {
874                    return Err(ProofCheckErrorKind::CongruenceChildMismatch {
875                        proof_id,
876                        child_lhs,
877                        base_child: children[*child_index],
878                        child_index: *child_index,
879                    }
880                    .into());
881                }
882
883                // Construct the expected new term by replacing the child
884                let expected_rhs_children: Vec<TermId> = children
885                    .iter()
886                    .enumerate()
887                    .map(
888                        |(i, &child)| {
889                            if i == *child_index { child_rhs } else { child }
890                        },
891                    )
892                    .collect();
893
894                let expected_rhs_id = self.term_dag.app(func_name, expected_rhs_children);
895
896                // Verify proof.rhs() matches expected
897                if proof.rhs() != expected_rhs_id {
898                    return Err(ProofCheckErrorKind::CongruenceResultMismatch {
899                        proof_id,
900                        proof_rhs: proof.rhs(),
901                        expected_rhs: expected_rhs_id,
902                    }
903                    .into());
904                }
905
906                // Verify proof.lhs() matches base_lhs
907                if proof.lhs() != base_lhs {
908                    return Err(ProofCheckErrorKind::CongruenceLhsMismatch { proof_id }.into());
909                }
910
911                Ok(Proposition::new(proof.lhs(), proof.rhs()))
912            }
913
914            Justification::ContainerNormalize { proof: inner_id } => {
915                // The sub-proof establishes `t1 = raw`; normalize `raw` to its
916                // canonical container form via the validator for its head.
917                let inner_prop = self.check_proof_with_context(*inner_id, program, ctx)?;
918                let raw = inner_prop.rhs;
919                let normalized = self.normalize_container(raw);
920                let lhs_ok = proof.lhs() == inner_prop.lhs;
921                if !lhs_ok || proof.rhs() != normalized {
922                    return Err(ProofCheckErrorKind::ContainerNormalizeMismatch {
923                        proof_id,
924                        raw,
925                        normalized,
926                        proof_rhs: proof.rhs(),
927                        lhs_ok,
928                    }
929                    .into());
930                }
931                Ok(Proposition::new(proof.lhs(), proof.rhs()))
932            }
933
934            Justification::Eval => {
935                // The `Eval` marker is the proof of a container side condition and
936                // is checked by re-evaluation in `check_side_condition` (driven by
937                // the rule body), never on its own. Reaching it here means it
938                // appeared outside a side condition, which is malformed.
939                Err(ProofCheckErrorKind::EvalOutsideSideCondition { proof_id }.into())
940            }
941        };
942
943        // Cache the result
944        if let Ok(ref prop) = result {
945            ctx.checked_proofs.insert(proof_id, prop.clone());
946        }
947
948        result
949    }

951    /// Check a container side condition by re-evaluating it against the rule
952    /// body, rather than against a premise proposition. An unbound side is the
953    /// side condition's output and is bound; otherwise both sides must evaluate
954    /// to the same container. Extends `subst` with any bound output.
955    fn check_side_condition(
956        &mut self,
957        fact: &ResolvedFact,
958        subst: &mut HashMap<String, TermId>,
959        rule_name: &str,
960    ) -> Result<(), ProofCheckError> {
961        let ResolvedFact::Eq(_, lhs, rhs) = fact else {
962            // `is_container_side_condition` only flags `Eq` facts.
963            return Ok(());
964        };
965        let lhs_val = self.eval_side(lhs, subst, rule_name)?;
966        let rhs_val = self.eval_side(rhs, subst, rule_name)?;
967        match (lhs, lhs_val, rhs, rhs_val) {
968            // One side is an unbound variable: it is the side condition's output.
969            (ResolvedExpr::Var(_, v), None, _, Some(val))
970            | (_, Some(val), ResolvedExpr::Var(_, v), None) => {
971                subst.insert(v.name.clone(), val);
972                Ok(())
973            }
974            // Both sides determined: they must be the same container.
975            (_, Some(l), _, Some(r)) => {
976                if l != r {
977                    return Err(ProofCheckErrorKind::SideConditionMismatch {
978                        rule_name: rule_name.to_string(),
979                        fact: format!("{fact}"),
980                        lhs: l,
981                        rhs: r,
982                    }
983                    .into());
984                }
985                Ok(())
986            }
987            _ => Err(ProofCheckErrorKind::SideConditionUnbound {
988                rule_name: rule_name.to_string(),
989                fact: format!("{fact}"),
990            }
991            .into()),
992        }
993    }

995    /// Evaluate one side of a side condition: an unbound variable yields `None`
996    /// (it is an output to bind), anything else is evaluated with the rule's
997    /// typed primitive validators.
998    fn eval_side(
999        &mut self,
1000        expr: &ResolvedExpr,
1001        subst: &HashMap<String, TermId>,
1002        rule_name: &str,
1003    ) -> Result<Option<TermId>, ProofCheckError> {
1004        match expr {
1005            ResolvedExpr::Var(_, v) => Ok(subst.get(&v.name).copied()),
1006            _ => {
1007                let (term, _) = eval_expr_with_subst(rule_name, expr, &mut self.term_dag, subst)?;
1008                Ok(Some(term))
1009            }
1010        }
1011    }

1013    /// Reject a rule-body fact that isn't in proof normal form: a primitive must
1014    /// not have a constructor/function argument, and a container-producing
1015    /// primitive must not appear nested (it must be its own side condition).
1016    fn assert_body_proof_normal_form(
1017        &self,
1018        fact: &ResolvedFact,
1019        rule_name: &str,
1020    ) -> Result<(), ProofCheckError> {
1021        fn check(expr: &ResolvedExpr, rule_name: &str) -> Result<(), ProofCheckError> {
1022            let ResolvedExpr::Call(_, head, args) = expr else {
1023                return Ok(());
1024            };
1025            for arg in args {
1026                match head {
1027                    ResolvedCall::Primitive(_)
1028                        if matches!(arg, ResolvedExpr::Call(_, ResolvedCall::Func(_), _)) =>
1029                    {
1030                        return Err(ProofCheckErrorKind::PrimitiveNonNormalArg {
1031                            rule_name: rule_name.to_string(),
1032                            arg: format!("{arg}"),
1033                        }
1034                        .into());
1035                    }
1036                    ResolvedCall::Func(_)
1037                        if matches!(
1038                            arg,
1039                            ResolvedExpr::Call(_, ResolvedCall::Primitive(p), _)
1040                                if p.output().is_eq_container_sort()
1041                        ) =>
1042                    {
1043                        return Err(ProofCheckErrorKind::ContainerPrimitiveNotSideCondition {
1044                            rule_name: rule_name.to_string(),
1045                            prim: format!("{arg}"),
1046                        }
1047                        .into());
1048                    }
1049                    _ => {}
1050                }
1051                check(arg, rule_name)?;
1052            }
1053            Ok(())
1054        }
1055        match fact {
1056            ResolvedFact::Eq(_, lhs, rhs) => {
1057                check(lhs, rule_name)?;
1058                check(rhs, rule_name)?;
1059            }
1060            ResolvedFact::Fact(expr) => check(expr, rule_name)?,
1061        }
1062        Ok(())
1063    }

1065    /// Check that a fact matches a proposition under a substitution
1066    fn check_fact_matches_proposition(
1067        &mut self,
1068        fact: &ResolvedFact,
1069        prop: &Proposition,
1070        subst_with_globals: &HashMap<String, TermId>,
1071        rule_name: &str,
1072    ) -> Result<(), ProofCheckError> {
1073        let (lhs, rhs) = (prop.lhs, prop.rhs);
1074        match fact {
1075            // proof normal form for functions: (= (f args...) v)
1076            // In the term representation, custom functions store output as last arg: f(args..., v)
1077            ResolvedFact::Eq(
1078                _,
1079                ResolvedExpr::Call(_, call @ ResolvedCall::Func(_), args),
1080                ResolvedExpr::Var(_, v),
1081            ) if call.is_custom_func() => {
1082                let name = call.name();
1083                // Get the output variable's term
1084                let var_term = subst_with_globals.get(&v.name).copied().ok_or_else(|| {
1085                    ProofCheckErrorKind::UnboundVariable {
1086                        rule_name: rule_name.to_string(),
1087                        variable: v.name.clone(),
1088                        available: subst_with_globals
1089                            .keys()
1090                            .cloned()
1091                            .collect::<Vec<_>>()
1092                            .join(", "),
1093                    }
1094                })?;
1095
1096                // Evaluate all the input arguments
1097                let mut arg_terms = Vec::new();
1098                for arg in args {
1099                    arg_terms.push(self.eval_expr_with_subst(
1100                        rule_name,
1101                        arg,
1102                        subst_with_globals,
1103                    )?);
1104                }
1105                // Add the output variable as the last argument
1106                arg_terms.push(var_term);
1107
1108                let expected_term_id = self.term_dag.app(name.to_owned(), arg_terms);
1109
1110                // The proposition should be a reflexive equality for this term
1111                if lhs != expected_term_id || rhs != expected_term_id {
1112                    return Err(ProofCheckErrorKind::FunctionFactMismatch {
1113                        rule_name: rule_name.to_string(),
1114                        expected: format_term(&self.term_dag, expected_term_id),
1115                        actual_lhs: format_term(&self.term_dag, lhs),
1116                        actual_rhs: format_term(&self.term_dag, rhs),
1117                    }
1118                    .into());
1119                }
1120
1121                Ok(())
1122            }
1123            ResolvedFact::Eq(_, lhs_expr, rhs_expr) => {
1124                let fact_lhs =
1125                    self.eval_expr_with_subst(rule_name, lhs_expr, subst_with_globals)?;
1126                let fact_rhs =
1127                    self.eval_expr_with_subst(rule_name, rhs_expr, subst_with_globals)?;
1128                if fact_lhs != lhs || fact_rhs != rhs {
1129                    return Err(ProofCheckErrorKind::EqualityFactMismatch {
1130                        rule_name: rule_name.to_string(),
1131                        fact: format!("{fact}"),
1132                        substituted_lhs: self.term_dag.to_string(fact_lhs),
1133                        substituted_rhs: self.term_dag.to_string(fact_rhs),
1134                        proven_lhs: format_term(&self.term_dag, lhs),
1135                        proven_rhs: format_term(&self.term_dag, rhs),
1136                    }
1137                    .into());
1138                }
1139
1140                Ok(())
1141            }
1142            // For a plain expr, the proof should have the form t1 = t2 where t2 matches the expr under substitution
1143            ResolvedFact::Fact(expr) => {
1144                let fact_term = self.eval_expr_with_subst(rule_name, expr, subst_with_globals)?;
1145
1146                if fact_term != rhs {
1147                    return Err(ProofCheckErrorKind::FactMismatch {
1148                        rule_name: rule_name.to_string(),
1149                        fact: format!("{fact}"),
1150                        substitution: format_substitution(&self.term_dag, subst_with_globals),
1151                        actual: format_term(&self.term_dag, rhs),
1152                        expected: format_term(&self.term_dag, fact_term),
1153                    }
1154                    .into());
1155                }
1156
1157                Ok(())
1158            }
1159        }
1160    }

1162    /// Evaluate an expression with a variable substitution
1163    fn eval_expr_with_subst(
1164        &mut self,
1165        rule_name: &str,
1166        expr: &ResolvedExpr,
1167        substitution: &HashMap<String, TermId>,
1168    ) -> Result<TermId, ProofCheckError> {
1169        match expr {
1170            ResolvedExpr::Lit(_, lit) => Ok(self.term_dag.lit(lit.clone())),
1171            ResolvedExpr::Var(_, var) => substitution.get(&var.name).copied().ok_or_else(|| {
1172                ProofCheckError::from(ProofCheckErrorKind::UnboundVariable {
1173                    rule_name: rule_name.to_string(),
1174                    variable: var.name.clone(),
1175                    available: substitution.keys().cloned().collect::<Vec<_>>().join(", "),
1176                })
1177            }),
1178            ResolvedExpr::Call(_, head, args) => {
1179                // Evaluate all arguments first
1180                let mut arg_terms = Vec::new();
1181                for arg in args {
1182                    arg_terms.push(self.eval_expr_with_subst(rule_name, arg, substitution)?);
1183                }
1184
1185                match head {
1186                    ResolvedCall::Primitive(prim) => {
1187                        // Use the validator to compute the primitive result
1188                        if let Some(validator) = prim.validator() {
1189                            let result =
1190                                validator(&mut self.term_dag, &arg_terms).ok_or_else(|| {
1191                                    ProofCheckErrorKind::PrimitiveValidatorFailed {
1192                                        function_name: prim.name().to_string(),
1193                                    }
1194                                })?;
1195                            Ok(result)
1196                        } else {
1197                            // No validator available - primitives without validators can't be checked in proofs
1198                            Err(ProofCheckErrorKind::PrimitiveNoValidator {
1199                                function_name: prim.name().to_string(),
1200                            }
1201                            .into())
1202                        }
1203                    }
1204                    ResolvedCall::Func(func) => {
1205                        match func.subtype {
1206                            FunctionSubtype::Constructor => {
1207                                Ok(self.term_dag.app(func.name.clone(), arg_terms))
1208                            }
1209                            FunctionSubtype::Custom => {
1210                                // Custom functions should not appear in proof normal form!
1211                                // They should be in the form (= (f args...) v) in the rule body
1212                                panic!(
1213                                    "Custom function {} should not appear in expression evaluation during proof checking. \
1214                                    Functions should be in proof normal form: (= (function args...) output_var)",
1215                                    func.name
1216                                );
1217                            }
1218                        }
1219                    }
1220                }
1221            }
1222        }
1223    }

1225    /// Check that a rule produces the claimed equality
1226    fn check_rule_produces_equality(
1227        &mut self,
1228        rule: &crate::ast::GenericRule<ResolvedCall, crate::ast::ResolvedVar>,
1229        substitution: &HashMap<String, TermId>,
1230        subst_with_globals: &HashMap<String, TermId>,
1231        claimed: &Proposition,
1232        rule_name: &str,
1233    ) -> Result<(), ProofCheckError> {
1234        // Use process_actions to get propositions from the rule head
1235        // Note: process_actions expects global variable bindings, but substitution
1236        // has the same structure, so we can pass it directly
1237        let action_refs: Vec<&GenericAction<ResolvedCall, crate::ast::ResolvedVar>> =
1238            rule.head.0.iter().collect();
1239        let bindings = subst_with_globals.clone();
1240        let action_ctx = process_actions(rule_name, bindings, &action_refs, &mut self.term_dag)?;
1241
1242        // Check if the claimed equality is in the propositions
1243        if action_ctx.propositions.contains(claimed) {
1244            return Ok(());
1245        }
1246
1247        Err(ProofCheckErrorKind::RuleHeadMismatch {
1248            rule_name: rule_name.to_string(),
1249            claimed_lhs: format_term(&self.term_dag, claimed.lhs()),
1250            claimed_rhs: format_term(&self.term_dag, claimed.rhs()),
1251            substitution: format_substitution(&self.term_dag, substitution),
1252        }
1253        .into())
1254    }
1255}
