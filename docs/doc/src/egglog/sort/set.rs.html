4#[derive(Clone, Debug, PartialEq, Eq, Hash)]
5pub struct SetContainer {
6    pub do_rebuild: bool,
7    pub data: BTreeSet<Value>,
8}
9
10impl ContainerValue for SetContainer {
11    fn rebuild_contents(&mut self, rebuilder: &dyn ValueRebuilder) -> bool {
12        if self.do_rebuild {
13            let mut xs: Vec<_> = self.data.iter().copied().collect();
14            let changed = rebuilder.rebuild_slice(&mut xs);
15            self.data = xs.into_iter().collect();
16            changed
17        } else {
18            false
19        }
20    }
21    fn iter(&self) -> impl Iterator<Item = Value> + '_ {
22        self.data.iter().copied()
23    }
24}

26/// The elements of a `(set-of e0 ...)` term as a Rust `BTreeSet` in AST
27/// order, matching `SetContainer`'s semantics; `None` for any other term.
28fn set_term_to_btreeset<'a>(termdag: &'a TermDag, term: TermId) -> Option<BTreeSet<OrdTerm<'a>>> {
29    match termdag.get(term) {
30        Term::App(head, children) if head == "set-of" => {
31            Some(children.iter().map(|c| termdag.ord_term(*c)).collect())
32        }
33        _ => None,
34    }
35}
36
37/// Flatten a set back to the element list of its canonical `(set-of ...)`
38/// term (sorted by AST order and deduplicated by construction).
39fn set_term_args(set: BTreeSet<OrdTerm<'_>>) -> Vec<TermId> {
40    set.into_iter().map(|e| e.id()).collect()
41}
42
43/// Canonicalize `elements` to the `(set-of e0 e1 ...)` term form: sorted by
44/// [`TermDag::ast_cmp`] and deduplicated, so proof checking can reproduce it.
45fn normalize_set_term(termdag: &mut TermDag, elements: &[TermId]) -> TermId {
46    let set: BTreeSet<_> = elements.iter().map(|e| termdag.ord_term(*e)).collect();
47    let elements = set_term_args(set);
48    termdag.app("set-of".into(), elements)
49}

84    fn make_sort(
85        typeinfo: &mut TypeInfo,
86        name: String,
87        args: &[Expr],
88        span: Span,
89    ) -> Result<ArcSort, TypeError> {
90        if let [Expr::Var(arg_span, e)] = args {
91            let e = typeinfo
92                .get_sort_by_name(e)
93                .ok_or(TypeError::UndefinedSort(e.clone(), arg_span.clone()))?;
94
95            let out = Self {
96                name,
97                element: e.clone(),
98            };
99            Ok(out.to_arcsort())
100        } else {
101            Err(TypeError::BadPresortArguments(
102                Self::presort_name().to_owned(),
103                span,
104            ))
105        }
106    }

139    fn register_primitives(&self, eg: &mut EGraph) {
140        let arc = self.clone().to_arcsort();
141
142        // Proof term form of a set: `(set-of e0 e1 ...)` sorted and
143        // deduplicated, matching `reconstruct_termdag`. Each validator
144        // round-trips through a Rust `BTreeSet` (see `set_term_to_btreeset`),
145        // so it evaluates set terms with `SetContainer`'s semantics.
146        let set_of_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
147            Some(normalize_set_term(termdag, args))
148        };
149        let set_empty_validator = |termdag: &mut TermDag, _args: &[TermId]| -> Option<TermId> {
150            Some(termdag.app("set-of".into(), vec![]))
151        };
152        let set_insert_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
153            let [set, value] = args else {
154                return None;
155            };
156            let mut set = set_term_to_btreeset(termdag, *set)?;
157            set.insert(termdag.ord_term(*value));
158            let elements = set_term_args(set);
159            Some(termdag.app("set-of".into(), elements))
160        };
161        let set_remove_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
162            let [set, value] = args else {
163                return None;
164            };
165            let mut set = set_term_to_btreeset(termdag, *set)?;
166            set.remove(&termdag.ord_term(*value));
167            let elements = set_term_args(set);
168            Some(termdag.app("set-of".into(), elements))
169        };
170        let set_length_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
171            let [set] = args else {
172                return None;
173            };
174            let len = set_term_to_btreeset(termdag, *set)?.len() as i64;
175            Some(termdag.lit(Literal::Int(len)))
176        };
177        let set_contains_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
178            let [set, value] = args else {
179                return None;
180            };
181            let contains = set_term_to_btreeset(termdag, *set)?.contains(&termdag.ord_term(*value));
182            contains.then(|| termdag.lit(Literal::Unit))
183        };
184        let set_not_contains_validator = |termdag: &mut TermDag,
185                                          args: &[TermId]|
186         -> Option<TermId> {
187            let [set, value] = args else {
188                return None;
189            };
190            let contains = set_term_to_btreeset(termdag, *set)?.contains(&termdag.ord_term(*value));
191            (!contains).then(|| termdag.lit(Literal::Unit))
192        };
193        let set_union_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
194            let [left, right] = args else {
195                return None;
196            };
197            let mut set = set_term_to_btreeset(termdag, *left)?;
198            set.extend(set_term_to_btreeset(termdag, *right)?);
199            let elements = set_term_args(set);
200            Some(termdag.app("set-of".into(), elements))
201        };
202        let set_diff_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
203            let [left, right] = args else {
204                return None;
205            };
206            let mut set = set_term_to_btreeset(termdag, *left)?;
207            let right = set_term_to_btreeset(termdag, *right)?;
208            set.retain(|e| !right.contains(e));
209            let elements = set_term_args(set);
210            Some(termdag.app("set-of".into(), elements))
211        };
212        let set_intersect_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
213            let [left, right] = args else {
214                return None;
215            };
216            let mut set = set_term_to_btreeset(termdag, *left)?;
217            let right = set_term_to_btreeset(termdag, *right)?;
218            set.retain(|e| right.contains(e));
219            let elements = set_term_args(set);
220            Some(termdag.app("set-of".into(), elements))
221        };
222
223        add_primitive_with_validator!(eg, "set-empty" = {self.clone(): SetSort} |                      | -> @SetContainer (arc) { SetContainer {
224            do_rebuild: self.ctx.is_eq_container_sort(),
225            data: BTreeSet::new()
226        } }, set_empty_validator);
227        add_primitive_with_validator!(eg, "set-of"    = {self.clone(): SetSort} [xs: # (self.element())] -> @SetContainer (arc) { SetContainer {
228            do_rebuild: self.ctx.is_eq_container_sort(),
229            data: xs.collect()
230        } }, set_of_validator);
231
232        // No validator: `set-get` indexes the runtime `BTreeSet<Value>` order,
233        // which terms cannot reproduce, so it is unsupported in proof mode.
234        add_primitive!(eg, "set-get" = |xs: @SetContainer (arc), i: i64| -?> # (self.element()) { xs.data.iter().nth(i as usize).copied() });
235        add_primitive_with_validator!(eg, "set-insert" = |mut xs: @SetContainer (arc), x: # (self.element())| -> @SetContainer (arc) {{ xs.data.insert( x); xs }}, set_insert_validator);
236        add_primitive_with_validator!(eg, "set-remove" = |mut xs: @SetContainer (arc), x: # (self.element())| -> @SetContainer (arc) {{ xs.data.remove(&x); xs }}, set_remove_validator);
237
238        add_primitive_with_validator!(eg, "set-length"       = |xs: @SetContainer (arc)| -> i64 { xs.data.len() as i64 }, set_length_validator);
239        add_primitive_with_validator!(eg, "set-contains"     = |xs: @SetContainer (arc), x: # (self.element())| -?> () { ( xs.data.contains(&x)).then_some(()) }, set_contains_validator);
240        add_primitive_with_validator!(eg, "set-not-contains" = |xs: @SetContainer (arc), x: # (self.element())| -?> () { (!xs.data.contains(&x)).then_some(()) }, set_not_contains_validator);
241
242        add_primitive_with_validator!(eg, "set-union"      = |mut xs: @SetContainer (arc), ys: @SetContainer (arc)| -> @SetContainer (arc) {{ xs.data.extend(ys.data);                  xs }}, set_union_validator);
243        add_primitive_with_validator!(eg, "set-diff"       = |mut xs: @SetContainer (arc), ys: @SetContainer (arc)| -> @SetContainer (arc) {{ xs.data.retain(|k| !ys.data.contains(k)); xs }}, set_diff_validator);
244        add_primitive_with_validator!(eg, "set-intersect"  = |mut xs: @SetContainer (arc), ys: @SetContainer (arc)| -> @SetContainer (arc) {{ xs.data.retain(|k|  ys.data.contains(k)); xs }}, set_intersect_validator);
245    }

253    ) -> TermId {
