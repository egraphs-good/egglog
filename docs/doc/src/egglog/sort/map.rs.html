4#[derive(Clone, Debug, PartialEq, Eq, Hash)]
5pub struct MapContainer {
6    do_rebuild_keys: bool,
7    do_rebuild_vals: bool,
8    pub data: BTreeMap<Value, Value>,
9}
10
11impl ContainerValue for MapContainer {
12    fn rebuild_contents(&mut self, rebuilder: &dyn ValueRebuilder) -> bool {
13        let mut changed = false;
14        if self.do_rebuild_keys {
15            self.data = self
16                .data
17                .iter()
18                .map(|(old, v)| {
19                    let new = rebuilder.rebuild_val(*old);
20                    changed |= *old != new;
21                    (new, *v)
22                })
23                .collect();
24        }
25        if self.do_rebuild_vals {
26            for old in self.data.values_mut() {
27                let new = rebuilder.rebuild_val(*old);
28                changed |= *old != new;
29                *old = new;
30            }
31        }
32        changed
33    }
34    fn iter(&self) -> impl Iterator<Item = Value> + '_ {
35        self.data.iter().flat_map(|(k, v)| [k, v]).copied()
36    }
37}

39/// The entries of a flat `(map-of k0 v0 ...)` term as a Rust `BTreeMap` in
40/// canonical key order, with `MapContainer`'s last-write-wins semantics on
41/// duplicate keys; `None` for any other term.
42fn map_term_to_btreemap<'a>(
43    termdag: &'a TermDag,
44    term_id: TermId,
45) -> Option<BTreeMap<OrdTerm<'a>, TermId>> {
46    match termdag.get(term_id) {
47        Term::App(head, args) if head == "map-of" => map_of_args_to_btreemap(termdag, args),
48        _ => None,
49    }
50}
51
52/// Alternating `[k0, v0, ...]` `map-of` arguments as a `BTreeMap` (see
53/// [`map_term_to_btreemap`]); `None` on odd arity.
54fn map_of_args_to_btreemap<'a>(
55    termdag: &'a TermDag,
56    args: &[TermId],
57) -> Option<BTreeMap<OrdTerm<'a>, TermId>> {
58    if !args.len().is_multiple_of(2) {
59        return None;
60    }
61    Some(
62        args.chunks_exact(2)
63            .map(|kv| (termdag.ord_term(kv[0]), kv[1]))
64            .collect(),
65    )
66}
67
68/// Flatten a map back to the `[k0, v0, k1, v1, ...]` argument list of its
69/// canonical `(map-of ...)` term (sorted by key order, deduplicated).
70fn map_term_args(map: BTreeMap<OrdTerm<'_>, TermId>) -> Vec<TermId> {
71    map.into_iter().flat_map(|(k, v)| [k.id(), v]).collect()
72}
73
74/// Canonicalize alternating `[k0, v0, ...]` arguments to the flat
75/// `(map-of ...)` term; `None` on odd arity.
76fn normalize_map_term(termdag: &mut TermDag, args: &[TermId]) -> Option<TermId> {
77    let flat = map_term_args(map_of_args_to_btreemap(termdag, args)?);
78    Some(termdag.app("map-of".to_string(), flat))
79}

81/// A map from a key type to a value type supporting these primitives:
82/// - `map-empty`
83/// - `map-insert`
84/// - `map-get`
85/// - `map-contains`
86/// - `map-not-contains`
87/// - `map-remove`
88/// - `map-length`
89#[derive(Clone, Debug)]
90pub struct MapSort {
91    name: String,
92    key: ArcSort,
93    value: ArcSort,
94}
95
96impl MapSort {
97    pub fn key(&self) -> ArcSort {
98        self.key.clone()
99    }
100
101    pub fn value(&self) -> ArcSort {
102        self.value.clone()
103    }
104}

124    fn make_sort(
125        typeinfo: &mut TypeInfo,
126        name: String,
127        args: &[Expr],
128        span: Span,
129    ) -> Result<ArcSort, TypeError> {
130        if let [Expr::Var(k_span, k), Expr::Var(v_span, v)] = args {
131            let k = typeinfo
132                .get_sort_by_name(k)
133                .ok_or(TypeError::UndefinedSort(k.clone(), k_span.clone()))?;
134            let v = typeinfo
135                .get_sort_by_name(v)
136                .ok_or(TypeError::UndefinedSort(v.clone(), v_span.clone()))?;
137
138            let out = Self {
139                name,
140                key: k.clone(),
141                value: v.clone(),
142            };
143            Ok(out.to_arcsort())
144        } else {
145            Err(TypeError::BadPresortArguments(
146                Self::presort_name().to_owned(),
147                span,
148            ))
149        }
150    }

153impl ContainerSort for MapSort {
154    type Container = MapContainer;
155
156    fn name(&self) -> &str {
157        &self.name
158    }
159
160    fn inner_sorts(&self) -> Vec<ArcSort> {
161        vec![self.key.clone(), self.value.clone()]
162    }
163
164    fn is_eq_container_sort(&self) -> bool {
165        self.key.is_eq_sort()
166            || self.value.is_eq_sort()
167            || self.key.is_eq_container_sort()
168            || self.value.is_eq_container_sort()
169    }
170
171    fn inner_values(
172        &self,
173        container_values: &ContainerValues,
174        value: Value,
175    ) -> Vec<(ArcSort, Value)> {
176        let val = container_values
177            .get_val::<MapContainer>(value)
178            .unwrap()
179            .clone();
180        val.data
181            .iter()
182            .flat_map(|(k, v)| [(self.key.clone(), *k), (self.value.clone(), *v)])
183            .collect()
184    }
185
186    fn register_primitives(&self, eg: &mut EGraph) {
187        let arc = self.clone().to_arcsort();
188
189        // The proof "term form" of a map is the flat `(map-of k0 v0 k1 v1 ...)`
190        // in canonical key order (like `set-of`/`vec-of`), matching
191        // `reconstruct_termdag`. Each validator round-trips through a Rust
192        // `BTreeMap` (see `map_term_to_btreemap`), so it evaluates map terms
193        // with `MapContainer`'s semantics; `None` for a malformed map term
194        // fails the proof.
195        let map_empty_validator = |termdag: &mut TermDag, _args: &[TermId]| -> Option<TermId> {
196            Some(termdag.app("map-of".into(), vec![]))
197        };
198        let map_insert_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
199            let [map, key, value] = args else {
200                return None;
201            };
202            let mut map = map_term_to_btreemap(termdag, *map)?;
203            map.insert(termdag.ord_term(*key), *value);
204            let flat = map_term_args(map);
205            Some(termdag.app("map-of".into(), flat))
206        };
207        let map_get_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
208            let [map, key] = args else { return None };
209            map_term_to_btreemap(termdag, *map)?
210                .get(&termdag.ord_term(*key))
211                .copied()
212        };
213        let map_length_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
214            let [map] = args else { return None };
215            let len = map_term_to_btreemap(termdag, *map)?.len() as i64;
216            Some(termdag.lit(Literal::Int(len)))
217        };
218        let map_contains_validator = |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
219            let [map, key] = args else { return None };
220            let contains =
221                map_term_to_btreemap(termdag, *map)?.contains_key(&termdag.ord_term(*key));
222            contains.then(|| termdag.lit(Literal::Unit))
223        };
224        let map_not_contains_validator =
225            |termdag: &mut TermDag, args: &[TermId]| -> Option<TermId> {
226                let [map, key] = args else { return None };
227                let contains =
228                    map_term_to_btreemap(termdag, *map)?.contains_key(&termdag.ord_term(*key));
229                (!contains).then(|| termdag.lit(Literal::Unit))
230            };
231
232        add_primitive_with_validator!(eg, "map-empty" = {self.clone(): MapSort} || -> @MapContainer (arc) { MapContainer {
233            do_rebuild_keys: self.ctx.key.is_eq_sort() || self.ctx.key.is_eq_container_sort(),
234            do_rebuild_vals: self.ctx.value.is_eq_sort() || self.ctx.value.is_eq_container_sort(),
235            data: BTreeMap::new()
236        } }, map_empty_validator);
237
238        // `map-of` is the flat constructor used as the canonical term form. It
239        // takes alternating key/value arguments, so it needs a custom type
240        // constraint rather than the `add_primitive!` macro.
241        eg.add_pure_primitive(
242            MapOf {
243                name: "map-of".to_string(),
244                map: arc.clone(),
245                key: self.key.clone(),
246                value: self.value.clone(),
247            },
248            Some(std::sync::Arc::new(normalize_map_term)),
249        );
250
251        add_primitive_with_validator!(eg, "map-get"    = |    xs: @MapContainer (arc), x: # (self.key())                     | -?> # (self.value()) { xs.data.get(&x).copied() }, map_get_validator);
252        add_primitive_with_validator!(eg, "map-insert" = |mut xs: @MapContainer (arc), x: # (self.key()), y: # (self.value())| -> @MapContainer (arc) {{ xs.data.insert(x, y); xs }}, map_insert_validator);
253        add_primitive!(eg, "map-remove" = |mut xs: @MapContainer (arc), x: # (self.key())                     | -> @MapContainer (arc) {{ xs.data.remove(&x);   xs }});
254
255        add_primitive_with_validator!(eg, "map-length"       = |xs: @MapContainer (arc)| -> i64 { xs.data.len() as i64 }, map_length_validator);
256        add_primitive_with_validator!(eg, "map-contains"     = |xs: @MapContainer (arc), x: # (self.key())| -?> () { ( xs.data.contains_key(&x)).then_some(()) }, map_contains_validator);
257        add_primitive_with_validator!(eg, "map-not-contains" = |xs: @MapContainer (arc), x: # (self.key())| -?> () { (!xs.data.contains_key(&x)).then_some(()) }, map_not_contains_validator);
258    }
259
260    fn reconstruct_termdag(
261        &self,
262        _container_values: &ContainerValues,
263        _value: Value,
264        termdag: &mut TermDag,
265        element_terms: Vec<TermId>,
266    ) -> TermId {
267        // Flat `(map-of k0 v0 k1 v1 ...)` in canonical key order, so proof
268        // checking can reproduce it from terms alone (and the rebuild proof's
269        // Congr indices are flat, like `set-of`/`vec-of`).
270        normalize_map_term(termdag, &element_terms).expect("map elements come in key/value pairs")
271    }
272
273    fn rebuild_container_normalizer(&self) -> Option<(String, PrimitiveValidator)> {
274        Some(("map-of".to_owned(), Arc::new(normalize_map_term)))
275    }
276
277    fn serialized_name(&self, _container_values: &ContainerValues, _: Value) -> String {
278        "map-of".to_owned()
279    }

282/// The flat `map-of` constructor: takes alternating key/value arguments and
283/// builds a map. Used as the canonical term form for maps (analogous to
284/// `set-of`/`vec-of`). Needs a custom type constraint because its arguments
285/// alternate between the key and value sorts.
286#[derive(Clone)]
287struct MapOf {
288    name: String,
289    map: ArcSort,
290    key: ArcSort,
291    value: ArcSort,
292}
293
294impl Primitive for MapOf {
295    fn name(&self) -> &str {
296        &self.name
297    }
298
299    fn get_type_constraints(&self, span: &Span) -> Box<dyn TypeConstraint> {
300        Box::new(MapOfTypeConstraint {
301            name: self.name.clone(),
302            key: self.key.clone(),
303            value: self.value.clone(),
304            map: self.map.clone(),
305            span: span.clone(),
306        })
307    }
308}
309
310impl PurePrim for MapOf {
311    fn apply<'a, 'db>(&self, mut state: PureState<'a, 'db>, args: &[Value]) -> Option<Value> {
312        let mut data = BTreeMap::new();
313        for chunk in args.chunks(2) {
314            if let [k, v] = chunk {
315                data.insert(*k, *v);
316            }
317        }
318        let mc = MapContainer {
319            do_rebuild_keys: self.key.is_eq_sort() || self.key.is_eq_container_sort(),
320            do_rebuild_vals: self.value.is_eq_sort() || self.value.is_eq_container_sort(),
321            data,
322        };
323        Some(state.register_container(mc))
324    }
325}

327/// Type constraint for [`MapOf`]: an even number of inputs alternating between
328/// the key and value sorts, producing the map sort.
329struct MapOfTypeConstraint {
330    name: String,
331    key: ArcSort,
332    value: ArcSort,
333    map: ArcSort,
334    span: Span,
335}
336
337impl TypeConstraint for MapOfTypeConstraint {
338    fn get(
339        &self,
340        arguments: &[AtomTerm],
341        _typeinfo: &TypeInfo,
342    ) -> Vec<Box<dyn Constraint<AtomTerm, ArcSort>>> {
343        let arity_mismatch = |expected: usize| {
344            vec![constraint::impossible(
345                constraint::ImpossibleConstraint::ArityMismatch {
346                    atom: Atom {
347                        span: self.span.clone(),
348                        head: self.name.clone(),
349                        args: arguments.to_vec(),
350                    },
351                    expected,
352                },
353            )]
354        };
355        let Some((out, inputs)) = arguments.split_last() else {
356            return arity_mismatch(1);
357        };
358        if inputs.len() % 2 != 0 {
359            return arity_mismatch(inputs.len() + 2);
360        }
361        let mut cs: Vec<Box<dyn Constraint<AtomTerm, ArcSort>>> =
362            vec![constraint::assign(out.clone(), self.map.clone())];
363        for (i, arg) in inputs.iter().enumerate() {
364            let sort = if i % 2 == 0 {
365                self.key.clone()
366            } else {
367                self.value.clone()
368            };
369            cs.push(constraint::assign(arg.clone(), sort));
370        }
371        cs
372    }
