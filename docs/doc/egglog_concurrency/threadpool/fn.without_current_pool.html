pub fn without_current_pool<F, R>(f: F) -> R
where
    F: FnOnce() -> R,
Expand description

Run f with no current thread pool installed on this thread.

This is useful for serial operations that must not inherit an ambient pool. The previous current pool, if any, is restored before this function returns or unwinds.

§Examples

use egglog_concurrency::{ThreadPool, current_num_threads, without_current_pool};

let pool = ThreadPool::new(2);
pool.install(|| {
    assert_eq!(current_num_threads(), 2);
    without_current_pool(|| {
        assert_eq!(current_num_threads(), 1);
    });
    assert_eq!(current_num_threads(), 2);
});
