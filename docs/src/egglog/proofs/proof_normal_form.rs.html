6/// Transforms queries into "proof normal form" by lifting subexpressions to the
7/// top level, so that every primitive is applied only to variables, literals, or
8/// other primitives. This is what lets the proof checker re-evaluate a primitive
9/// directly (see `check_side_condition`) instead of needing a proof for each of a
10/// primitive's arguments — the arguments are already bound elsewhere.
11///
12/// 1. A custom function call becomes its own top-level fact:
13///    `(= (lower-bound a b) c)`.
14/// 2. A constructor or function argument of a primitive is lifted to a fresh
15///    variable: `(!= a (Const 0))` becomes `(= (Const 0) v)`, `(!= a v)`.
16/// 3. A container-producing primitive is lifted out of any constructor into its
17///    own side condition: `(WrapVec (vec-of e))` becomes `(= (vec-of e) v)`,
18///    `(WrapVec v)`. Its proof is a contentless marker, which can't ride a
19///    congruence step under the constructor.
20pub(crate) fn proof_form(
21    prog: Vec<ResolvedNCommand>,
22    fresh: &mut SymbolGen,
23) -> Vec<ResolvedNCommand> {
24    prog.into_iter()
25        .map(|cmd| proof_form_cmd(cmd, fresh))
26        .collect()
27}
28
29fn proof_form_cmd(cmd: ResolvedNCommand, fresh: &mut SymbolGen) -> ResolvedNCommand {
30    cmd.visit_queries(&mut |query| {
31        let mut new_query = vec![];
32        for fact in query {
33            let rewritten = proof_form_fact(fact, &mut new_query, fresh);
34            new_query.push(rewritten);
35        }
36        new_query
37    })
38}

40fn proof_form_fact(
41    fact: ResolvedFact,
42    res: &mut Vec<ResolvedFact>,
43    fresh: &mut SymbolGen,
44) -> ResolvedFact {
45    match fact {
46        ResolvedFact::Eq(
47            span,
48            ResolvedExpr::Call(
49                span2,
50                head @ ResolvedCall::Func(FuncType {
51                    subtype: FunctionSubtype::Custom,
52                    ..
53                }),
54                args,
55            ),
56            ResolvedExpr::Var(span3, v),
57        ) => {
58            let mut new_args = vec![];
59            for arg in args {
60                new_args.push(proof_form_expr(arg, res, fresh));
61            }
62            ResolvedFact::Eq(
63                span,
64                ResolvedExpr::Call(span2, head, new_args),
65                ResolvedExpr::Var(span3, v),
66            )
67        }
68        GenericFact::Eq(span, generic_expr, generic_expr2) => GenericFact::Eq(
69            span,
70            proof_form_expr(generic_expr, res, fresh),
71            proof_form_expr(generic_expr2, res, fresh),
72        ),
73        GenericFact::Fact(generic_expr) => {
74            GenericFact::Fact(proof_form_expr(generic_expr, res, fresh))
75        }
76    }
77}

79fn proof_form_expr(
80    fact: ResolvedExpr,
81    res: &mut Vec<ResolvedFact>,
82    fresh: &mut SymbolGen,
83) -> ResolvedExpr {
84    match fact {
85        ref fact @ ResolvedExpr::Call(
86            ref span,
87            ref head @ ResolvedCall::Func(FuncType {
88                subtype: FunctionSubtype::Custom,
89                ref output,
90                ..
91            }),
92            ref args,
93        ) => {
94            // bind this to a new variable
95            let new_args = args
96                .iter()
97                .map(|expr| proof_form_expr(expr.clone(), res, fresh))
98                .collect();
99            let resolved = GenericExpr::Var(
100                span.clone(),
101                ResolvedVar {
102                    name: fresh.fresh("n"),
103                    sort: output.clone(),
104                    is_global_ref: false,
105                },
106            );
107            res.push(ResolvedFact::Eq(
108                span.clone(),
109                ResolvedExpr::Call(span.clone(), head.clone(), new_args),
110                resolved.clone(),
111            ));
112            log::warn!(
113                "Input program not in proof normal form! All function calls must be top-level in query.
114            Original fact: {fact}
115            New top level fact: {}
116            Replace with new variable {}
117                ",
118                res.last().unwrap(),
119                resolved
120            );
121
122            resolved
123        }
124        ResolvedExpr::Call(span, head @ ResolvedCall::Primitive(_), args) => {
125            // For primitives, extract any constructor/custom function call arguments
126            // into separate facts with fresh variables. Other primitives can stay inline.
127            let mut new_args = vec![];
128            for arg in args {
129                match arg {
130                    // If the argument is a constructor or custom function call, extract it
131                    // (but allow other primitives to stay inline)
132                    ref arg_expr @ ResolvedExpr::Call(
133                        ref arg_span,
134                        ResolvedCall::Func(FuncType { ref output, .. }),
135                        ref inner_args,
136                    ) => {
137                        // First recursively normalize the inner arguments
138                        let normalized_inner_args: Vec<_> = inner_args
139                            .iter()
140                            .map(|e| proof_form_expr(e.clone(), res, fresh))
141                            .collect();
142
143                        // Create a fresh variable for this constructor call
144                        let fresh_var = GenericExpr::Var(
145                            arg_span.clone(),
146                            ResolvedVar {
147                                name: fresh.fresh("v"),
148                                sort: output.clone(),
149                                is_global_ref: false,
150                            },
151                        );
152
153                        // Add an equality fact binding the constructor to the fresh variable
154                        res.push(ResolvedFact::Eq(
155                            arg_span.clone(),
156                            ResolvedExpr::Call(
157                                arg_span.clone(),
158                                match arg_expr {
159                                    ResolvedExpr::Call(_, call, _) => call.clone(),
160                                    _ => unreachable!(),
161                                },
162                                normalized_inner_args,
163                            ),
164                            fresh_var.clone(),
165                        ));
166
167                        new_args.push(fresh_var);
168                    }
169                    // Otherwise just recursively normalize
170                    other => {
171                        new_args.push(proof_form_expr(other, res, fresh));
172                    }
173                }
174            }
175            ResolvedExpr::Call(span, head, new_args)
176        }
177        ResolvedExpr::Call(span, head, args) => {
178            // `head` is a constructor here (custom functions and primitives are
179            // matched above). A container-producing primitive can't sit under a
180            // constructor in proof normal form — it has no anchored proof and
181            // can't ride a congruence step — so lift such an argument into its
182            // own side-condition binding `(= (prim ...) v)` and pass `v`.
183            let mut new_args = vec![];
184            for arg in args {
185                let normalized = proof_form_expr(arg, res, fresh);
186                let lift = matches!(
187                    &normalized,
188                    ResolvedExpr::Call(_, ResolvedCall::Primitive(p), _)
189                        if p.output().is_eq_container_sort()
190                );
191                if lift {
192                    let (arg_span, sort) = match &normalized {
193                        ResolvedExpr::Call(s, ResolvedCall::Primitive(p), _) => {
194                            (s.clone(), p.output().clone())
195                        }
196                        _ => unreachable!(),
197                    };
198                    let fresh_var = GenericExpr::Var(
199                        arg_span.clone(),
200                        ResolvedVar {
201                            name: fresh.fresh("v"),
202                            sort,
203                            is_global_ref: false,
204                        },
205                    );
206                    res.push(ResolvedFact::Eq(arg_span, normalized, fresh_var.clone()));
207                    new_args.push(fresh_var);
208                } else {
209                    new_args.push(normalized);
210                }
211            }
212            ResolvedExpr::Call(span, head, new_args)
213        }
214        ResolvedExpr::Lit(..) | ResolvedExpr::Var(..) => fact,
215    }
