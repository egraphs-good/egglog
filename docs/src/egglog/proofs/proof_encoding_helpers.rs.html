52/// Packages proof information for instrumenting actions.

57    Fiat,
58    Proof(String),                 // existing proof

339    /// Header string for proof encoding, defining sorts and constructors.
340    /// Correspondings to [`RawProof`] in the Rust code.
341    pub(crate) fn proof_header(&mut self) -> String {
342        let mut to_ast_constructors = Vec::new();
343        // need to build a Ast{lit} for each lit sort in self
344        for sort_name in self.egraph.type_info.sorts.keys().clone() {
345            if !self
346                .proof_names()
347                .sort_to_ast_constructor
348                .contains_key(sort_name)
349            {
350                let ast_constructor = self
351                    .egraph
352                    .parser
353                    .symbol_gen
354                    .fresh(&format!("Ast{sort_name}"));
355                self.egraph
356                    .proof_state
357                    .proof_names
358                    .sort_to_ast_constructor
359                    .insert(sort_name.clone(), ast_constructor.clone());
360                to_ast_constructors.push(format!(
361                    "(constructor {ast_constructor} ({sort_name} ) {} :internal-hidden)",
362                    self.proof_names().ast_sort
363                ));
364            }
365        }
366        let to_ast_str = to_ast_constructors.join("\n");
367
368        let EncodingNames {
369            ref proof_list_sort,
370            ref ast_sort,
371            ref proof_datatype,
372            ref fiat_constructor,
373            ref rule_constructor,
374            ref merge_fn_constructor,
375            ref eq_trans_constructor,
376            ref eq_sym_constructor,
377            ref congr_constructor,
378            ref container_normalize_constructor,
379            ref eval_constructor,
380            ref pcons,
381            ref pnil,
382            ..
383        } = *self.proof_names();
384
385        format!(
386            "
387(sort {proof_list_sort})
388(sort {ast_sort}) ;; wrap sorts in this for proofs
389;; The proof datatype records the global proof constructor names so container
390;; rebuild can recover them on re-parse (see ContainerRebuildSpec).
391(sort {proof_datatype} :internal-proof-names {congr_constructor} {eq_trans_constructor} {eq_sym_constructor} {container_normalize_constructor})
392
393(constructor {pcons} ({proof_datatype} {proof_list_sort}) {proof_list_sort} :internal-hidden)
394(constructor {pnil} () {proof_list_sort} :internal-hidden)
395
396{to_ast_str}
397
398;; Fiat justification for globals and primitives, gives two terms t1 = t2 for the proposition being justified
399(constructor {fiat_constructor} ({ast_sort} {ast_sort}) {proof_datatype} :internal-hidden)
400;; name of rule, one proof per fact in the query, proposition being proven t1 = t2
401(constructor {rule_constructor} (String {proof_list_sort} {ast_sort} {ast_sort}) {proof_datatype} :internal-hidden)
402
403;; merge function justification- name of function and two proofs for the two terms being merged,
404;; and the proposition being justified t = t
405(constructor {merge_fn_constructor} (String {proof_datatype} {proof_datatype} {ast_sort}) {proof_datatype} :internal-hidden)
406
407;; transitivity of equality proofs
408(constructor {eq_trans_constructor} ({proof_datatype} {proof_datatype}) {proof_datatype} :internal-hidden)
409
410;; symmetry of equality proofs
411(constructor  {eq_sym_constructor} ({proof_datatype}) {proof_datatype} :internal-hidden)
412;; given a proof that t1 = f(..., ci, ...)
413;; and the child index i of ci in the term f(..., ci, ...)
414;; and a proof that ci = c2,
415;; produces a justification that t1 = f(..., c2, ...)
416(constructor  {congr_constructor} ({proof_datatype} i64 {proof_datatype}) {proof_datatype} :internal-hidden)
417
418;; given a proof that t1 = c, where c is a container term, produces a proof that
419;; t1 = normalize(c) (the container's canonicalization: sort/dedup for sets,
420;; last-write-wins for maps, sort for multisets)
421(constructor  {container_normalize_constructor} ({proof_datatype}) {proof_datatype} :internal-hidden)
422
423;; marks the proof of a container side condition. Carries nothing: the side
424;; condition is re-evaluated against the rule body when checked.
425(constructor  {eval_constructor} () {proof_datatype} :internal-hidden)
426                "
427        )
428    }

431/// Reads a file and checks that its commands support the proof encoding.
432pub fn file_supports_proofs(path: &Path) -> bool {
433    let contents = match std::fs::read_to_string(path) {
434        Ok(contents) => contents,
435        Err(_) => return false,
436    };
437
438    let canonical = match std::fs::canonicalize(path) {
439        Ok(canonical) => canonical,
440        Err(_) => return false,
441    };
442
443    let mut egraph = EGraph::default();
444    let filename = canonical.to_string_lossy().into_owned();
445    let desugared = match egraph.resolve_program(Some(filename.clone()), &contents) {
446        Ok(commands) => commands,
447        Err(_) => return false,
448    };
449
450    program_supports_proofs(&desugared, &egraph.type_info)
451}

453/// Reasons why a command doesn't support proof encoding
454#[derive(Debug, Clone, thiserror::Error)]
455pub enum ProofEncodingUnsupportedReason {
456    #[error("primitive operation lacks a validator function")]
457    PrimitiveWithoutValidator,
458    #[error(
459        "action contains a function lookup. Finding the output of a function is only supported in queries."
460    )]
461    FunctionLookupInAction,
462    #[error(
463        "a container constructed in the query (a container-producing primitive result) is used in the actions. A query-built container is a side condition with no carryable proof, so it cannot be carried into an action."
464    )]
465    ContainerCreatedInQueryUsedInAction,
466    #[error(
467        "sort has a presort (custom sort container implementation). Custom sorts are not supported by proof encoding."
468    )]
469    SortWithPresort,
470    #[error(
471        "sort has a :internal-uf annotation. The :internal-uf annotation is used internally by term encoding and cannot be specified manually in proof mode."
472    )]
473    SortWithUfAnnotation,
474    #[error(
475        "sort has a :internal-proof-func annotation. The :internal-proof-func annotation is used internally by proof encoding and cannot be specified manually in proof mode."
476    )]
477    SortWithProofFuncAnnotation,
478    #[error("user-defined commands are not supported.")]
479    UserDefinedCommand,
480    #[error("input commands are not supported.")]
481    InputCommand,
482    #[error("missing merge function. All functions need to specify a :merge function.")]
483    NoMergeOnNonGlobalFunction,
484    #[error(
485        "let binding with a primitive in the body. For silly internal reasons, we don't support primitive bindings for proofs at the moment, sorry."
486    )]
487    LetBindingWithNonEqSort,
488    #[error(
489        "rule uses `:unsafe-seminaive`. Arbitrary RHS database reads are not representable by the term/proof encoding."
490    )]
491    UnsafeSeminaive,
492    #[error(
493        "rule uses `:naive` with an eq-sort primitive in the body. Proof encoding can only look up proofs for primitive eq-sort fact results under seminaive-safe query evaluation."
494    )]
495    NaiveEqSortPrimitiveFact,
496}

498/// Checks whether a desugared program supports proof encoding.
499pub fn program_supports_proofs(commands: &[ResolvedCommand], type_info: &TypeInfo) -> bool {
500    for command in commands {
501        if command_supports_proof_encoding(command, type_info).is_err() {
502            return false;
503        }
504    }
505    true

508/// Recursively check if all primitives in an expression have validators
509fn expr_primitives_have_validators(expr: &ResolvedExpr) -> bool {
510    use crate::ast::GenericExpr;
511    use crate::core::ResolvedCall;
512
513    let mut all_valid = true;
514    expr.walk(
515        &mut |e| {
516            if let GenericExpr::Call(_, ResolvedCall::Primitive(prim), _) = e
517                && prim.validator().is_none()
518            {
519                all_valid = false;
520            }
521        },
522        &mut |_| {},
523    );
524    all_valid
525}

527/// Check if an action contains non-global function lookups in any of its expressions
528fn action_has_function_lookup(action: &ResolvedAction, type_info: &TypeInfo) -> bool {
529    let mut has_lookup = false;
530    action.clone().visit_exprs(&mut |expr| {
531        if type_info.expr_has_function_lookup(&expr).is_some() {
532            has_lookup = true;
533        }
534        expr
535    });
536    has_lookup
537}

539/// Check if a fact contains a primitive expression whose result needs a stored term proof.
540fn fact_has_eq_sort_primitive_result(fact: &ResolvedFact) -> bool {
541    let mut has_eq_sort_primitive = false;
542    fact.clone().visit_exprs(&mut |expr| {
543        if let ResolvedExpr::Call(_, ResolvedCall::Primitive(prim), _) = &expr
544            && (prim.output().is_eq_sort() || prim.output().is_eq_container_sort())
545        {
546            has_eq_sort_primitive = true;
547        }
548        expr
549    });
550    has_eq_sort_primitive
551}

553/// Checks whether a resolved command supports proof encoding.
554/// Returns Ok(()) if supported, or Err with the reason if not.
555pub(crate) fn command_supports_proof_encoding(
556    command: &ResolvedCommand,
557    type_info: &TypeInfo,
558) -> Result<(), ProofEncodingUnsupportedReason> {
559    // `:unsafe-seminaive` rules perform arbitrary reads against the live
560    // database; the term/proof encoding can't represent that.
561    if let crate::ast::GenericCommand::Rule { rule } = command
562        && rule.eval_mode == crate::ast::RuleEvalMode::UnsafeSeminaive
563    {
564        return Err(ProofEncodingUnsupportedReason::UnsafeSeminaive);
565    }
566    if let crate::ast::GenericCommand::Rule { rule } = command
567        && rule.eval_mode == crate::ast::RuleEvalMode::Naive
568        && rule.body.iter().any(fact_has_eq_sort_primitive_result)
569    {
570        return Err(ProofEncodingUnsupportedReason::NaiveEqSortPrimitiveFact);
571    }
572    // Check all expressions for primitives without validators
573    let mut all_primitives_have_validators = true;
574    command.clone().visit_exprs(&mut |expr| {
575        if !expr_primitives_have_validators(&expr) {
576            all_primitives_have_validators = false;
577        }
578        expr
579    });
580
581    if !all_primitives_have_validators {
582        return Err(ProofEncodingUnsupportedReason::PrimitiveWithoutValidator);
583    }
584
585    // Check actions (not queries) for function lookups
586    // Egglog supports lookups in actions at the global level, but not in proofs mode
587    // (global function calls are allowed - they get desugared to constructors)
588    let mut has_function_lookup_in_action = false;
589    command.clone().visit_actions(&mut |action| {
590        has_function_lookup_in_action |= action_has_function_lookup(&action, type_info);
591        action
592    });
593
594    if has_function_lookup_in_action {
595        return Err(ProofEncodingUnsupportedReason::FunctionLookupInAction);
596    }
597
598    // A container built by a primitive in the query is a side condition with no
599    // carryable proof, so it can't be used in an action. Reject a rule that binds
600    // such a container to a variable used in its actions.
601    if let GenericCommand::Rule { rule } = command {
602        let mut constructed: Vec<String> = Vec::new();
603        for fact in &rule.body {
604            if let ResolvedFact::Eq(_, lhs, rhs) = fact {
605                for (var_side, call_side) in [(lhs, rhs), (rhs, lhs)] {
606                    if let ResolvedExpr::Var(_, v) = var_side
607                        && let ResolvedExpr::Call(_, ResolvedCall::Primitive(prim), _) = call_side
608                        && prim.output().is_eq_container_sort()
609                    {
610                        constructed.push(v.name.clone());
611                    }
612                }
613            }
614        }
615        if !constructed.is_empty() {
616            let mut used_in_action = false;
617            for action in &rule.head.0 {
618                action.clone().visit_exprs(&mut |expr| {
619                    expr.walk(
620                        &mut |e| {
621                            if let ResolvedExpr::Var(_, v) = e
622                                && constructed.contains(&v.name)
623                            {
624                                used_in_action = true;
625                            }
626                        },
627                        &mut |_| {},
628                    );
629                    expr
630                });
631            }
632            if used_in_action {
633                return Err(ProofEncodingUnsupportedReason::ContainerCreatedInQueryUsedInAction);
634            }
635        }
636    }
637
638    // Now check command-specific constraints
639    match command {
640        GenericCommand::Sort {
641            name,
642            presort_and_args: Some(_),
643            ..
644        } => type_info
645            .get_sort_by_name(name)
646            .filter(|sort| sort.is_container_sort())
647            .map(|_| ())
648            .ok_or(ProofEncodingUnsupportedReason::SortWithPresort),
649        GenericCommand::Sort { uf: Some(_), .. } => {
650            Err(ProofEncodingUnsupportedReason::SortWithUfAnnotation)
651        }
652        GenericCommand::Sort {
653            proof_func: Some(_),
654            ..
655        } => Err(ProofEncodingUnsupportedReason::SortWithProofFuncAnnotation),
656        GenericCommand::UserDefined(..) => Err(ProofEncodingUnsupportedReason::UserDefinedCommand),
657        GenericCommand::Input { .. } => Err(ProofEncodingUnsupportedReason::InputCommand),
658        // Extract commands can't have non-global function lookups
659        // because instrument_action_expr doesn't support them
660        // (global function calls are fine - they get desugared to constructors)
661        GenericCommand::Extract(_, expr, variants) => {
662            if type_info.expr_has_function_lookup(expr).is_some()
663                || type_info.expr_has_function_lookup(variants).is_some()
664            {
665                Err(ProofEncodingUnsupportedReason::FunctionLookupInAction)
666            } else {
667                Ok(())
668            }
669        }
670        // no-merge on a non-global function
671        // To add support: https://github.com/egraphs-good/egglog/issues/774
672        GenericCommand::Function {
673            merge: None, name, ..
674        } => {
675            if type_info.is_global(name) {
676                Ok(())
677            } else {
678                Err(ProofEncodingUnsupportedReason::NoMergeOnNonGlobalFunction)
679            }
680        }
681        // let binding with non-eq sort not supported by proof_global_desugar
682        ResolvedCommand::Action(ResolvedAction::Let(_, _, expr)) => {
683            // let binding with non-eq sort not supported by proof_global_desugar
684            // we detect as setting something that is no-merge to a primitive not supported (global primitive binding)
685            if expr.output_type().is_eq_sort() {
686                Ok(())
687            } else {
688                Err(ProofEncodingUnsupportedReason::LetBindingWithNonEqSort)
689            }
690        }
691        // After global desugar it may look like this
692        ResolvedCommand::Action(ResolvedAction::Set(_span, head, _children, expr)) => {
693            if !type_info.is_global(head.name()) || expr.output_type().is_eq_sort() {
694                Ok(())
695            } else {
696                Err(ProofEncodingUnsupportedReason::LetBindingWithNonEqSort)
697            }
698        }
699        // (fail <cmd>) must still satisfy the proof encoding constraints on its inner command,
700        // e.g. (fail (let x 3)) should report the unsupported inner command rather than
701        // bypassing the guard and panicking later in proof global removal.
702        GenericCommand::Fail(_span, inner) => {
703            command_supports_proof_encoding(inner.as_ref(), type_info)
704        }
705        _ => Ok(()),
706    }
707}
