19/// Read-only information about the e-graph handed to a scheduler callback.
20pub struct SchedulerContext<'a> {
21    /// The e-graph the rules run on, for read-only queries such as
22    /// [`EGraph::get_size`], [`EGraph::total_size`], and [`EGraph::num_nodes`].
23    pub egraph: &'a EGraph,
24    num_nodes: &'a Cell<Option<usize>>,
25}
26
27impl SchedulerContext<'_> {
28    /// Return the current number of e-nodes. The count is computed only if the
29    /// scheduler asks for it, then cached until an action or rebuild may change
30    /// it.
31    /// Actions chosen for an earlier rule have run before the next callback.
32    /// Rebuilding is deferred, however, so merge functions may still increase
33    /// or decrease this count before the iteration finishes. `can_stop`
34    /// receives the final count after actions and rebuilding.
35    pub fn num_nodes(&self) -> usize {
36        match self.num_nodes.get() {
37            Some(num_nodes) => num_nodes,
38            None => {
39                let num_nodes = self.egraph.num_nodes();
40                self.num_nodes.set(Some(num_nodes));
41                num_nodes
42            }
43        }
44    }
45}

47/// A scheduler decides which matches to be applied for a rule.
48///
49/// The matches that are not chosen in this iteration will be delayed
50/// to the next iteration.
51pub trait Scheduler: dyn_clone::DynClone + Send + Sync {
52    /// Whether or not the rules can be considered as saturated once no database
53    /// changes were made in the current iteration.
54    ///
55    /// This is only called when the runner is otherwise saturated.
56    /// Default implementation just returns `true`.
57    fn can_stop(&mut self, ctx: &SchedulerContext<'_>, rules: &[&str], ruleset: &str) -> bool {
58        let _ = (ctx, rules, ruleset);
59        true
60    }
61
62    /// Filter the matches for a rule.
63    ///
64    /// Chosen matches are applied (actions run, database flushed) before the
65    /// next rule's `filter_matches` is called, so `ctx.num_nodes()` reflects
66    /// earlier decisions. Queries are still collected together before any
67    /// callback.
68    ///
69    /// Return `true` if the scheduler's next run of the rule should feed
70    /// `filter_matches` with a new iteration of matches.
71    fn filter_matches(
72        &mut self,
73        ctx: &SchedulerContext<'_>,
74        rule: &str,
75        ruleset: &str,
76        matches: &mut Matches,
77    ) -> bool;
78}

82/// A collection of matches produced by a rule.
83/// The user can choose which matches to be fired.
84pub struct Matches {
85    matches: Vec<Value>,
86    chosen: Vec<usize>,
87    vars: Vec<ResolvedVar>,
88    /// Width of each stored tuple in `matches`. This is `vars.len()` for an
89    /// ordinary rule. A rule whose head references no variables would otherwise
90    /// have zero-width tuples, making its match count unrecoverable; for those we
91    /// collect a single unit marker per match, so the width is 1 while `vars` is
92    /// empty.
93    tuple_width: usize,
94    all_chosen: bool,
95}
96
97/// A match is a tuple of values corresponding to the variables in a rule.
98/// It allows you to retrieve the value corresponding to a variable in the match.
99pub struct Match<'a> {
100    values: &'a [Value],
101    vars: &'a [ResolvedVar],
102}
103
104impl Match<'_> {
105    /// Get the value corresponding a variable in this match.
106    pub fn get_value(&self, var: &str) -> Value {
107        let idx = self.vars.iter().position(|v| v.name == var).unwrap();
108        self.values[idx]
109    }
110}
111
112impl Matches {
113    fn new(matches: Vec<Value>, vars: Vec<ResolvedVar>) -> Self {
114        // Variable-free rules collect one unit marker per match (see
115        // `SchedulerRuleInfo::new`), so each stored tuple is one value wide even
116        // though there are no variables.
117        let tuple_width = vars.len().max(1);
118        assert!(matches.len().is_multiple_of(tuple_width));
119        Self {
120            matches,
121            vars,
122            tuple_width,
123            chosen: Vec::new(),
124            all_chosen: false,
125        }
126    }
127
128    /// The number of matches in total.
129    pub fn match_size(&self) -> usize {
130        self.matches.len() / self.tuple_width
131    }
132
133    /// The length of a tuple.
134    pub fn tuple_len(&self) -> usize {
135        self.vars.len()
136    }
137
138    /// Get `idx`-th match.
139    pub fn get_match(&self, idx: usize) -> Match<'_> {
140        Match {
141            values: &self.matches[idx * self.tuple_len()..(idx + 1) * self.tuple_len()],
142            vars: &self.vars,
143        }
144    }
145
146    /// Pick the match at `idx` to be fired.
147    pub fn choose(&mut self, idx: usize) {
148        self.chosen.push(idx);
149    }
150
151    /// Pick all matches to be fired.
152    ///
153    /// This is more efficient than calling `choose` for each match.
154    pub fn choose_all(&mut self) {
155        self.all_chosen = true;
156    }
157
158    /// Whether at least one match has been chosen to be fired.
159    pub fn any_chosen(&self) -> bool {
160        (self.all_chosen && self.match_size() > 0) || !self.chosen.is_empty()
161    }
162
163    /// Apply the chosen matches and return the residual matches.
164    fn instantiate(
165        mut self,
166        state: &mut ExecutionState<'_>,
167        table_action: &TableAction,
168    ) -> Vec<Value> {
169        // Width of the stored tuples (1 for variable-free rules, see `new`) versus
170        // the number of variable columns actually written into the `decided` table.
171        // For a variable-free rule the stored unit marker is dropped and only the
172        // trailing unit is inserted, producing the single `(unit)` row that the
173        // action rule fires on.
174        let tuple_width = self.tuple_width;
175        let var_len = self.vars.len();
176        let unit = state.base_values().get(());
177
178        if self.all_chosen {
179            for row in self.matches.chunks(tuple_width) {
180                table_action.insert(
181                    state,
182                    row[..var_len].iter().cloned().chain(std::iter::once(unit)),
183                );
184            }
185            vec![]
186        } else {
187            for idx in self.chosen.iter() {
188                let row = &self.matches[idx * tuple_width..(idx + 1) * tuple_width];
189                table_action.insert(
190                    state,
191                    row[..var_len].iter().cloned().chain(std::iter::once(unit)),
192                );
193            }
194
195            // swap remove the chosen matches
196            self.chosen.sort_unstable();
197            self.chosen.dedup();
198            let mut p = self.match_size();
199            for c in self.chosen.into_iter().rev() {
200                // It's important to decrement `p` first, because otherwise it might underflow when
201                // matches are exhausted.
202                p -= 1;
203                if c != p {
204                    let idx_c = c * tuple_width;
205                    let idx_p = p * tuple_width;
206                    for i in 0..tuple_width {
207                        self.matches.swap(idx_c + i, idx_p + i);
208                    }
209                }
210            }
211            self.matches.truncate(p * tuple_width);
212
213            self.matches
214        }
215    }
216}

237    /// Runs a ruleset for one iteration using the given scheduler.
238    ///
239    /// If a rule action fails, effects completed before the error are preserved
240    /// and its decided worklist rows are cleared. A successful recovery rebuild
241    /// leaves the e-graph canonical for a later call. The rows are not retained
242    /// for retry, although a later query may rediscover the same logical match.
243    pub fn step_rules_with_scheduler(
244        &mut self,
245        scheduler_id: SchedulerId,
246        ruleset: &str,
247    ) -> Result<RunReport, Error> {
248        fn collect_rules<'a>(
249            ruleset: &str,
250            rulesets: &'a IndexMap<String, Ruleset>,
251            ids: &mut Vec<(String, &'a CompiledRule)>,
252        ) -> Result<(), Error> {
253            let Some(r) = rulesets.get(ruleset) else {
254                return Err(Error::BackendError(format!("no such ruleset: {ruleset}")));
255            };
256            match r {
257                Ruleset::Rules(rules) => {
258                    for (rule_name, rule) in rules.iter() {
259                        ids.push((rule_name.clone(), rule));
260                    }
261                }
262                Ruleset::Combined(sub_rulesets) => {
263                    for sub_ruleset in sub_rulesets {
264                        collect_rules(sub_ruleset, rulesets, ids)?;
265                    }
266                }
267            }
268            Ok(())
269        }
270
271        let mut rules = Vec::new();
272        let rulesets = std::mem::take(&mut self.rulesets);
273        let collected = collect_rules(ruleset, &rulesets, &mut rules);
274        // Restore `rulesets` before propagating any error so the EGraph is not
275        // left with its rulesets taken out.
276        if let Err(e) = collected {
277            self.rulesets = rulesets;
278            return Err(e);
279        }
280        let mut schedulers = std::mem::take(&mut self.schedulers);
281
282        // `rulesets` and `schedulers` are now taken out of `self`. The body below
283        // has several fallible steps (rule compilation, `run_rules`), so run it in
284        // a closure and restore both fields afterward no matter how it exits.
285        // Otherwise an early error would leave the EGraph with empty rulesets and
286        // schedulers.
287        let result = (|| -> Result<RunReport, Error> {
288            // Step 1: build all the query/action rules and worklist if have not already
289            let record = &mut schedulers[scheduler_id];
290            for (id, rule) in rules.iter() {
291                if !record.rule_info.contains_key(id) {
292                    let info = SchedulerRuleInfo::new(self, rule, id)?;
293                    record.rule_info.insert((*id).to_owned(), info);
294                }
295            }
296
297            // Step 2: run all the queries for one iteration
298            let query_rules = rules
299                .iter()
300                .filter_map(|(rule_id, _rule)| {
301                    let rule_info = record.rule_info.get(rule_id).unwrap();
302
303                    if rule_info.should_seek {
304                        Some(rule_info.query_rule)
305                    } else {
306                        None
307                    }
308                })
309                .collect::<Vec<_>>();
310
311            let query_iter_report = self
312                .backend
313                .run_rules(&query_rules, Some(&self.type_info))
314                .map_err(|e| Error::BackendError(e.to_string()))?;
315
316            // Steps 3 and 4: choose and immediately apply each rule's matches.
317            // Rebuilding remains deferred until every rule has run.
318            let mut action_iteration_report = IterationReport::default();
319            let mut any_applied = false;
320            let num_nodes = Cell::new(None);
321            for (rule_id, _rule) in rules.iter() {
322                let rule_info = record.rule_info.get_mut(rule_id).unwrap();
323                let ctx = SchedulerContext {
324                    egraph: self,
325                    num_nodes: &num_nodes,
326                };
327                let mut inserted = false;
328                self.backend
329                    .with_execution_state(Some(&self.type_info), |state| {
330                        let matches: Vec<Value> =
331                            std::mem::take(rule_info.matches.lock().unwrap().as_mut());
332                        let mut matches = Matches::new(matches, rule_info.free_vars.clone());
333                        rule_info.should_seek =
334                            record
335                                .scheduler
336                                .filter_matches(&ctx, rule_id, ruleset, &mut matches);
337                        inserted = matches.any_chosen();
338                        let table_action = TableAction::new(&self.backend, rule_info.decided);
339                        let residual = matches.instantiate(state, &table_action);
340                        // A naive query re-finds all residuals when the scheduler asks
341                        // for a fresh query. Otherwise they remain available for the
342                        // next callback; seminaive queries only add fresh matches.
343                        *rule_info.matches.lock().unwrap() =
344                            if rule_info.seminaive || !rule_info.should_seek {
345                                residual
346                            } else {
347                                Vec::new()
348                            };
349                    });
350                if inserted {
351                    self.backend.flush_updates_no_rebuild();
352                    any_applied = true;
353                    let action_result = self
354                        .backend
355                        .run_rules_no_rebuild(&[rule_info.action_rule], Some(&self.type_info));
356                    num_nodes.set(None);
357                    let rule_report = match action_result {
358                        Ok(report) => report,
359                        Err(action_error) => {
360                            // Keep effects that completed before the error, but
361                            // consume this batch before restoring canonical tables.
362                            self.backend.clear_table(rule_info.decided);
363                            if let Err(rebuild_error) = self.backend.rebuild_now() {
364                                return Err(Error::BackendError(format!(
365                                    "{action_error}; rebuilding after the failed action also failed: {rebuild_error}"
366                                )));
367                            }
368                            return Err(Error::BackendError(action_error.to_string()));
369                        }
370                    };
371                    let aggregate = &mut action_iteration_report.rule_set_report;
372                    let current = rule_report.rule_set_report;
373                    aggregate.changed |= current.changed;
374                    aggregate.search_and_apply_time += current.search_and_apply_time;
375                    aggregate.merge_time += current.merge_time;
376                    for (rule, reports) in current.rule_reports {
377                        aggregate
378                            .rule_reports
379                            .entry(rule)
380                            .or_default()
381                            .extend(reports);
382                    }
383                }
384            }
385            if any_applied {
386                action_iteration_report.rebuild_time = self
387                    .backend
388                    .rebuild_now()
389                    .map_err(|e| Error::BackendError(e.to_string()))?;
390                num_nodes.set(None);
391            }
392            let mut action_report = RunReport::singleton(ruleset, action_iteration_report);
393
394            // Step 5: combine the reports
395            let mut query_report = RunReport::singleton(ruleset, query_iter_report);
396
397            // query matches don't count
398            query_report.updated = false;
399            query_report.num_matches_per_rule.clear();
400            // Scheduler state should not count as database progress. Instead it
401            // determines whether a no-op iteration can be treated as fully stopped.
402            action_report.can_stop = !action_report.updated && {
403                let rule_ids = rules.iter().map(|(id, _)| id.as_str()).collect::<Vec<_>>();
404                let ctx = SchedulerContext {
405                    egraph: self,
406                    num_nodes: &num_nodes,
407                };
408                record.scheduler.can_stop(&ctx, &rule_ids, ruleset)
409            };
410
411            query_report.union(action_report);
412
413            Ok(query_report)
414        })();
415
416        self.rulesets = rulesets;
417        self.schedulers = schedulers;
418
419        result
420    }
421}

429/// To enable scheduling without modifying the backend,

432/// (rule (worklist vars false) (action ... (delete (worklist vars false))))
433#[derive(Clone)]
434struct SchedulerRuleInfo {
435    matches: Arc<Mutex<Vec<Value>>>,
436    should_seek: bool,
437    decided: FunctionId,
438    query_rule: RuleId,
439    action_rule: RuleId,
440    free_vars: Vec<ResolvedVar>,
441    /// Whether this rule's scheduler query uses delta evaluation.
442    seminaive: bool,
443}

470impl SchedulerRuleInfo {
471    fn new(
472        egraph: &mut EGraph,
473        rule: &CompiledRule,
474        name: &str,
475    ) -> Result<SchedulerRuleInfo, Error> {
476        let free_vars = rule
477            .core
478            .head
479            .get_free_vars()
480            .into_iter()
481            .collect::<Vec<_>>();
482        let unit_type = egraph.backend.base_values().get_ty::<()>();
483        let unit = egraph.backend.base_values().get(());
484        let unit_entry = egraph.backend.base_value_constant(());
485
486        let matches = Arc::new(Mutex::new(Vec::new()));
487        let collect_matches = egraph
488            .backend
489            .register_external_func(Box::new(CollectMatches::new(matches.clone())));
490        let schema = free_vars
491            .iter()
492            .map(|v| v.sort.column_ty(&egraph.backend))
493            .chain(std::iter::once(ColumnTy::Base(unit_type)))
494            .collect();
495        // This table is registered for primitive access, so its name must not
496        // shadow a user table in the name-indexed registry.
497        let decided_name = egraph.parser.symbol_gen.fresh("scheduler_decided");
498        let decided = egraph.backend.add_table(FunctionConfig {
499            schema,
500            default: DefaultVal::Const(unit),
501            merge: MergeFn::AssertEq,
502            name: decided_name,
503            can_subsume: false,
504        });
505
506        // Step 1: rebuild the query with the same evaluation mode and planner
507        // options as the ordinary compiled rule.
508        let mut qrule = egraph.backend.new_rule(name, rule.seminaive);
509        qrule.set_no_decomp(rule.no_decomp);
510        let mut qrule_builder = BackendRule::new(
511            qrule,
512            &egraph.functions,
513            &egraph.type_info,
514            rule.requires_read_context,
515        );
516        qrule_builder.query(&rule.core.body, rule.include_subsumed)?;
517        let mut entries = free_vars
518            .iter()
519            .map(|fv| qrule_builder.entry(&GenericAtomTerm::Var(span!(), fv.clone())))
520            .collect::<Vec<_>>();
521        // A rule whose head references no variables would otherwise collect empty
522        // tuples, leaving the scheduler unable to tell whether the query matched and
523        // so never applying its actions. Collect a single unit marker per match so
524        // the match count is recoverable.
525        if entries.is_empty() {
526            entries.push(unit_entry.clone());
527        }
528        let _var = qrule_builder.rb.call_external_func(
529            collect_matches,
530            &entries,
531            ColumnTy::Base(unit_type),
532            || "collect_matches".to_string(),
533        );
534        let qrule_id = qrule_builder.build();
535
536        // Step 2: build the action rule
537        let mut arule_builder = BackendRule::new(
538            egraph.backend.new_rule(name, false),
539            &egraph.functions,
540            &egraph.type_info,
541            rule.requires_read_context,
542        );
543        let mut entries = free_vars
544            .iter()
545            .map(|fv| arule_builder.entry(&GenericAtomTerm::Var(span!(), fv.clone())))
546            .collect::<Vec<_>>();
547        entries.push(unit_entry);
548        arule_builder
549            .rb
550            .query_table(decided, &entries, None)
551            .unwrap();
552        arule_builder.actions(&rule.core.head)?;
553        // Remove the entry as it's now done
554        entries.pop();
555        arule_builder.rb.remove(decided, &entries);
556        let arule_id = arule_builder.build();
557
558        Ok(SchedulerRuleInfo {
559            free_vars,
560            query_rule: qrule_id,
561            action_rule: arule_id,
562            matches,
563            decided,
564            should_seek: true,
565            seminaive: rule.seminaive,
566        })
567    }

603        let input = r#"

607        (run-schedule (saturate (run)))

620            let table_size = egraph.get_size("S");

627            let rule_name: Arc<str> = "test-rule".into();
628            let expected = if expected_matches == 0 {
629                vec![]
630            } else {
631                vec![(&rule_name, &expected_matches)]
632            };
633            assert_eq!(
634                report.num_matches_per_rule.iter().collect::<Vec<_>>(),
635                expected
636            );
637
638            // Because of semi-naive, the exact rules that are run are more than just `test-rule`
639            assert!(
640                report
641                    .search_and_apply_time_per_rule
642                    .keys()
643                    .all(|k| k.starts_with("test-rule"))

647                [&"test".into()]

649            assert_eq!(
650                report

652                    .keys()
653                    .collect::<Vec<_>>(),
654                [&"test".into()]
655            );
656
657            if report.can_stop {
658                break;
659            }
660        }
661
662        assert_eq!(iter, 12);

902    /// A scheduler that stops choosing matches once the e-graph holds
903    /// `limit` e-nodes.
904    #[derive(Clone)]
905    struct NodeLimitScheduler {
906        limit: usize,
907    }
908
909    impl Scheduler for NodeLimitScheduler {
910        fn can_stop(
911            &mut self,
912            ctx: &SchedulerContext<'_>,
913            _rules: &[&str],
914            _ruleset: &str,
915        ) -> bool {
916            ctx.num_nodes() >= self.limit
917        }
918
919        fn filter_matches(
920            &mut self,
921            ctx: &SchedulerContext<'_>,
922            _rule: &str,
923            _ruleset: &str,
924            matches: &mut Matches,
925        ) -> bool {
926            if ctx.num_nodes() < self.limit {
927                matches.choose_all();
928                true
929            } else {
930                false
931            }
932        }
933    }

935    #[test]
936    fn test_scheduler_sees_egraph_size() {
937        let mut egraph = EGraph::default();
938        let scheduler_id = egraph.add_scheduler(Box::new(NodeLimitScheduler { limit: 10 }));
939        // Each firing adds one `Num` e-node; `depth` rows (base-sort output)
940        // and `seen` rows (a relation, i.e. a constructor over a fresh
941        // non-unionable sort) are analysis data and must not count towards
942        // `num_nodes`.
943        let input = r#"
944        (ruleset grow)
945        (datatype Math (Num i64))
946        (function depth (Math) i64 :no-merge)
947        (relation seen (Math))
948        (Num 0)
949        (rule ((= e (Num i)) (< i 100))
950              ((Num (+ i 1)) (set (depth e) i) (seen e))
951              :ruleset grow :name "grow")
952        "#;
953        egraph.parse_and_run_program(None, input).unwrap();
954        for _ in 0..20 {
955            let report = egraph
956                .step_rules_with_scheduler(scheduler_id, "grow")
957                .unwrap();
958            if report.can_stop {
959                break;
960            }
961        }
962        // One match per iteration, so the scheduler stops exactly at the limit.
963        assert_eq!(egraph.get_size("Num"), 10);
964        assert_eq!(egraph.num_nodes(), 10);
965        // `depth` and `seen` rows exist but only count towards `total_size`.
966        assert!(egraph.get_size("depth") > 0);
967        assert!(egraph.get_size("seen") > 0);
968        assert_eq!(
969            egraph.total_size(),
970            egraph.num_nodes() + egraph.get_size("depth") + egraph.get_size("seen")
971        );
972    }

1003    /// A scheduler that records the e-graph size it observes at each
1004    /// `filter_matches` call.
1005    #[derive(Clone)]
1006    struct SizeProbeScheduler {
1007        sizes_seen: Arc<Mutex<Vec<(String, usize)>>>,
1008    }
1009
1010    impl Scheduler for SizeProbeScheduler {
1011        fn filter_matches(
1012            &mut self,
1013            ctx: &SchedulerContext<'_>,
1014            rule: &str,
1015            _ruleset: &str,
1016            matches: &mut Matches,
1017        ) -> bool {
1018            self.sizes_seen
1019                .lock()
1020                .unwrap()
1021                .push((rule.to_owned(), ctx.num_nodes()));
1022            matches.choose_all();
1023            true
1024        }
1025    }

1028    fn test_filter_matches_sees_fresh_sizes() {
1029        let mut egraph = EGraph::default();
1030        let sizes_seen = Arc::new(Mutex::new(Vec::new()));
1031        let scheduler_id = egraph.add_scheduler(Box::new(SizeProbeScheduler {
1032            sizes_seen: sizes_seen.clone(),
1033        }));
1034        // "a-grow" adds one Num per iteration; "b-watch" only matches. Since
1035        // each rule's matches are applied before the next rule is consulted,
1036        // "b-watch" must observe "a-grow"'s new node within the same iteration.
1037        let input = r#"
1038        (ruleset t)
1039        (datatype Math (Num i64))
1040        (Num 0)
1041        (rule ((= e (Num i)) (< i 10)) ((union e (Num (+ i 1)))) :ruleset t :name "a-grow")
1042        (rule ((Num i)) ((Num i)) :ruleset t :name "b-watch")
1043        "#;
1044        egraph.parse_and_run_program(None, input).unwrap();
1045        let report = egraph.step_rules_with_scheduler(scheduler_id, "t").unwrap();
1046
1047        assert_eq!(
1048            *sizes_seen.lock().unwrap(),
1049            vec![("a-grow".to_owned(), 1), ("b-watch".to_owned(), 2)]
1050        );
1051        assert_eq!(egraph.get_size("Num"), 2);
1052        assert_eq!(
1053            report.iterations.len(),
1054            2,
1055            "one scheduler step has one query report and one action report"
1056        );
1057        assert_eq!(report.iterations[1].rule_reports().len(), 2);
