21/// Generate `n_cols` column blocks of `(Value, RowId)` pairs. Each block is RowId-ascending
22/// (matching a table scan) with values drawn from `0..distinct`, so equal-value runs occur.
23///
24/// Returns the concatenated pairs and the block bounds: block `b` is `[bounds[b], bounds[b + 1])`.
25pub fn gen_blocks(
26    n_rows: usize,
27    n_cols: usize,
28    distinct: u32,
29    seed: u64,
30) -> (Vec<(Value, RowId)>, Vec<usize>) {
31    let mut rng = StdRng::seed_from_u64(seed);
32    let mut pairs = Vec::with_capacity(n_rows * n_cols);
33    let mut bounds = vec![0usize];
34    for _ in 0..n_cols {
35        for r in 0..n_rows {
36            pairs.push((
37                Value::new(rng.random_range(0..distinct)),
38                RowId::from_usize(r),
39            ));
40        }
41        bounds.push(pairs.len());
42    }
43    (pairs, bounds)
44}

46/// The custom sort/merge from `rebuild_full`: radix-sort each block by value, then merge the
47/// blocks with dedup (a single block skips the merge).
48pub fn sort_merge_custom(mut pairs: Vec<(Value, RowId)>, bounds: &[usize]) -> Vec<(Value, RowId)> {
49    let widest = bounds.windows(2).map(|w| w[1] - w[0]).max().unwrap_or(0);
50    let mut scratch = vec![(Value::new_const(0), RowId::new_const(0)); widest];
51    for b in 0..bounds.len() - 1 {
52        radix_sort_slice_by_value(&mut pairs[bounds[b]..bounds[b + 1]], &mut scratch);
53    }
54    if bounds.len() <= 2 {
55        pairs
56    } else {
57        merge_sorted_blocks_dedup(pairs, bounds)
58    }
59}

61/// The standard-library analogue of [`sort_merge_custom`]: sort the whole concatenation by
62/// `(Value, RowId)`, then drop duplicates iff `dedup` is set.
63///
64/// Pass `dedup = false` for a single column and `true` for several, mirroring what
65/// `sort_merge_custom` does: a single block has one pair per row, so every `RowId` is distinct
66/// and no `(Value, RowId)` pair can repeat -- deduping there is a wasted pass that would unfairly
67/// handicap this side of the comparison. Only multiple columns can put the same value (hence the
68/// same pair) on one row.
69pub fn sort_merge_std(mut pairs: Vec<(Value, RowId)>, dedup: bool) -> Vec<(Value, RowId)> {
70    pairs.sort_unstable();
71    if dedup {
72        pairs.dedup();
73    }
74    pairs
75}

77/// Run `f` with an index thread pool of `n` threads installed, so parallel construction actually
78/// forks: `ColumnIndex` derives both its shard count and worker-pool size from the ambient pool.
79pub fn with_threads<R>(n: usize, f: impl FnOnce() -> R) -> R {
80    ThreadPool::new(n.max(1)).install(f)
81}

83/// A random table set up to (re)build a `ColumnIndex` over its value columns (`1..=n_val_cols`;
84/// column 0 is a unique key so no rows merge).
85pub struct IndexInput {
86    table: WrappedTable,
87    cols: Vec<ColumnId>,
88}
89
90impl IndexInput {
91    pub fn random(n_rows: usize, n_val_cols: usize, distinct: u32, seed: u64) -> Self {
92        let mut rng = StdRng::seed_from_u64(seed);
93        let n_cols = n_val_cols + 1;
94        let mut table = SortedWritesTable::new(
95            1,
96            n_cols,
97            None,
98            vec![],
99            Box::new(|_, _old, new, out: &mut Vec<Value>| {
100                out.extend_from_slice(new);
101                false
102            }),
103        );
104        {
105            let mut buf = table.new_buffer();
106            let mut row = vec![Value::new(0); n_cols];
107            for i in 0..n_rows {
108                row[0] = Value::from_usize(i);
109                for cell in row.iter_mut().skip(1) {
110                    *cell = Value::new(rng.random_range(0..distinct));
111                }
112                buf.stage_insert(&row);
113            }
114        }
115        let db = Database::default();
116        db.with_execution_state(|es| {
117            table.merge(es);
118        });
119        IndexInput {
120            table: WrappedTable::new(table),
121            cols: (1..n_cols).map(ColumnId::from_usize).collect(),
122        }
123    }
124
125    /// Build the index with the serial radix-sort + merge path.
126    pub fn build_serial(&self) {
127        let mut ci = ColumnIndex::new();
128        ci.rebuild_full(&self.cols, self.table.as_ref(), self.table.all().as_ref());
129        std::hint::black_box(&ci);
130    }
131
132    /// Build the index with the parallel sharded path. Call inside [`with_threads`].
133    pub fn build_parallel(&self) {
134        let mut ci = ColumnIndex::new();
135        ci.merge_parallel(&self.cols, self.table.as_ref(), self.table.all().as_ref());
136        std::hint::black_box(&ci);
137    }
138}
