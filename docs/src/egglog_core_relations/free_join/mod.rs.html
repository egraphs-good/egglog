180impl TableInfo {
181    /// Intern the family that indexes `column` after applying `constraints`.
182    fn successor_family(&mut self, column: ColumnId, constraints: &[Constraint]) -> FamilyId {
183        let next = FamilyId::from_usize(self.successor_families.len());
184        *self
185            .successor_families
186            .entry((column, canonical_constraints(constraints)))
187            .or_insert(next)
188    }
189
190    /// The child storage of every trie node shared below this table's roots:
191    /// one family per interned successor, or a leaf if no plan indexes below
192    /// them. Plans are built before a rule set runs, so the count is fixed for
193    /// the whole run.
194    pub(crate) fn shared_child_shape(&self) -> ChildShape {
195        if self.successor_families.is_empty() {
196            ChildShape::Leaf
197        } else {
198            ChildShape::Dynamic {
199                families: self.successor_families.len(),
200            }
201        }
202    }

207    }

222impl Clone for TableInfo {
223    fn clone(&self) -> Self {
224        fn deep_clone_map<K: Clone + std::hash::Hash + Eq, TI: IndexBase + Clone>(
225            map: &IndexCatalog<K, Arc<ResettableOnceLock<Index<TI>>>>,
226            table: WrappedTableRef,
227        ) -> IndexCatalog<K, Arc<ResettableOnceLock<Index<TI>>>> {
228            map.map(|table_ref| {
229                let (k, v) = table_ref;
230                let v: Index<TI> = v
231                    .get_or_update(|index| {
232                        index.refresh(table);
233                    })
234                    .clone();
235                (k.clone(), Arc::new(ResettableOnceLock::new(v)))
236            })
237        }
238        TableInfo {
239            identity: self.identity,
240            name: self.name.clone(),
241            spec: self.spec.clone(),
242            table: self.table.dyn_clone(),
243            indexes: deep_clone_map(&self.indexes, self.table.as_ref()),
244            column_indexes: deep_clone_map(&self.column_indexes, self.table.as_ref()),
245            successor_families: self.successor_families.clone(),
246        }
247    }

253/// External functions allow external callers to manipulate database state in
254/// near-arbitrary ways.
255///
256/// This is a useful, if low-level, interface for extending this database with
257/// functionality and state not built into the core model.
258pub trait ExternalFunction: dyn_clone::DynClone + Send + Sync {
259    /// Invoke the function with mutable access to the database. If a value is
260    /// not returned, halt the execution of the current rule.
261    fn invoke(&self, state: &mut ExecutionState, args: &[Value]) -> Option<Value>;
262}

264/// Automatically generate an `ExternalFunction` implementation from a function.
265pub fn make_external_func<
266    F: Fn(&mut ExecutionState, &[Value]) -> Option<Value> + Clone + Send + Sync,
267>(
268    f: F,
269) -> impl ExternalFunction {
270    #[derive(Clone)]
271    struct Wrapped<F>(F);
272    impl<F> ExternalFunction for Wrapped<F>
273    where
274        F: Fn(&mut ExecutionState, &[Value]) -> Option<Value> + Clone + Send + Sync,
275    {
276        fn invoke(&self, state: &mut ExecutionState, args: &[Value]) -> Option<Value> {
277            (self.0)(state, args)
278        }
279    }
280    Wrapped(f)
281}

283/// A vectorized variant of [`ExternalFunction::invoke`] to avoid repeated dynamic dispatch.
284pub(crate) fn invoke_batch(
285    this: &dyn ExternalFunction,
286    state: &mut ExecutionState,
287    mask: &mut Mask,
288    bindings: &mut Bindings,
289    args: &[QueryEntry],
290    out_var: Variable,
291) {
292    let pool: Pool<Vec<Value>> = with_pool_set(|ps| ps.get_pool());
293    let mut out = pool.get();
294    out.reserve(mask.len());
295    for_each_binding_with_mask!(mask, args, bindings, |iter| {
296        iter.fill_vec(&mut out, Value::stale, |_, args| {
297            this.invoke(state, args.as_slice())
298        });
299    });
300    bindings.insert(out_var, &out);
301}
302
303/// A variant of [`invoke_batch`] that overwrites the output variable,
304/// rather than assigning all new values.
305///
306/// *Panics* This method will panic if `out_var` doesn't already have an appropriately-sized
307/// vector bound in `bindings`.
308pub(crate) fn invoke_batch_assign(
309    this: &dyn ExternalFunction,
310    state: &mut ExecutionState,
311    mask: &mut Mask,
312    bindings: &mut Bindings,
313    args: &[QueryEntry],
314    out_var: Variable,
315) {
316    let mut out = bindings.take(out_var).expect("out_var must be bound");
317    for_each_binding_with_mask!(mask, args, bindings, |iter| {
318        iter.assign_vec_and_retain(&mut out.vals, |_, args| this.invoke(state, &args))
319    });
320    bindings.replace(out);
321}

329// Reservable counters give each execution state a disjoint range of values.
330// Values are then handed out by advancing the state's local `range`, avoiding
331// a shared atomic operation per value. Dropping the reservation returns its
332// unused suffix to `recycled`, and future states reuse those suffixes before
333// advancing the atomic high-water mark. A reservation size of one retains the
334// exact increment/read behavior needed by observable counters.
335struct Counter {
336    next: AtomicUsize,
337    reservation_size: usize,
338    recycled: SegQueue<std::ops::Range<usize>>,
339}
340
341impl Counter {
342    fn new(reservation_size: usize) -> Self {
343        assert!(reservation_size > 0);
344        Self {
345            next: AtomicUsize::new(0),
346            reservation_size,
347            recycled: SegQueue::new(),
348        }
349    }
350
351    fn take_reservation(&self) -> std::ops::Range<usize> {
352        if self.reservation_size == 1 {
353            let start = self.next.fetch_add(1, Ordering::Release);
354            return start..start + 1;
355        }
356        self.recycled.pop().unwrap_or_else(|| {
357            let start = self
358                .next
359                .fetch_add(self.reservation_size, Ordering::Release);
360            start..start + self.reservation_size
361        })
362    }
363}
364
365pub(crate) struct CounterReservation {
366    counter: Arc<Counter>,
367    range: std::ops::Range<usize>,
368}
369
370impl CounterReservation {
371    fn new(counter: Arc<Counter>) -> Self {
372        let range = counter.take_reservation();
373        Self { counter, range }
374    }
375
376    pub(crate) fn next(&mut self) -> usize {
377        if self.range.start == self.range.end {
378            self.range = self.counter.take_reservation();
379        }
380        let result = self.range.start;
381        self.range.start += 1;
382        result
383    }
384}
385
386impl Drop for CounterReservation {
387    fn drop(&mut self) {
388        if !self.range.is_empty() {
389            self.counter.recycled.push(self.range.clone());
390        }
391    }
392}

394#[derive(Default)]
395pub(crate) struct Counters(DenseIdMap<CounterId, Arc<Counter>>);
396
397impl Clone for Counters {
398    fn clone(&self) -> Counters {
399        let mut map = DenseIdMap::new();
400        for (k, v) in self.0.iter() {
401            // NB: we may want to experiment with Ordering::Relaxed here.
402            let cloned = Counter {
403                next: AtomicUsize::new(v.next.load(Ordering::SeqCst)),
404                reservation_size: v.reservation_size,
405                // The high-water mark already includes every recycled range,
406                // so omitting the free list is safe and avoids sharing
407                // reservations between independent database snapshots.
408                recycled: SegQueue::new(),
409            };
410            map.insert(k, Arc::new(cloned));
411        }
412        Counters(map)
413    }
414}
415
416impl Counters {
417    pub(crate) fn read(&self, ctr: CounterId) -> usize {
418        self.0[ctr].next.load(Ordering::Acquire)
419    }
420    pub(crate) fn inc(&self, ctr: CounterId) -> usize {
421        // We synchronize with `read_counter` but not with other increments.
422        // NB: we may want to experiment with Ordering::Relaxed here.
423        self.0[ctr].next.fetch_add(1, Ordering::Release)
424    }
425    pub(crate) fn take_reservation(&self, ctr: CounterId) -> CounterReservation {
426        CounterReservation::new(Arc::clone(&self.0[ctr]))
427    }
428}

508    /// Apply the value-level rebuild encoded by `func_id` to all the tables in `to_rebuild`.
509    ///
510    /// The native [`Table::apply_rebuild`] method takes a `next_ts` argument for filling in new
511    /// values in a table like [`crate::SortedWritesTable`] where values in a certain column need
512    /// to be inserted in sorted order; the `next_ts` argument to this method is passed to
513    /// `apply_rebuild` for this purpose.
514    pub fn apply_rebuild(
515        &mut self,
516        func_id: TableId,
517        to_rebuild: &[TableId],
518        next_ts: Value,
519    ) -> bool {
520        let func = self.tables.take(func_id).unwrap();
521        self.run_on_tables(to_rebuild, |_, info, view| {
522            info.table.apply_rebuild(
523                func_id,
524                &func.table,
525                next_ts,
526                &mut ExecutionState::new(*view, Default::default()),
527            )
528        });
529        self.tables.insert(func_id, func);
530        self.merge_all()
531    }

533    pub fn refresh_rows_for_values(
534        &mut self,
535        to_refresh: &[TableId],
536        dirty_ids: &[Value],
537        next_ts: Value,
538    ) -> bool {
539        if dirty_ids.is_empty() {
540            return false;
541        }
542        // This is the follow-up for `ContainerRebuildSummary::dirty_ids()`.
543        // These ids changed semantics without changing identity, so parent
544        // rows can become newly matchable without getting an ordinary table
545        // delta.
546        //
547        // It must run after ordinary table rebuild, which already handles
548        // changed-id cases by rewriting parent rows to the new id.
549        self.run_on_tables(to_refresh, |_, info, _| {
550            info.table.refresh_rows_for_values(dirty_ids, next_ts)
551        });
552        self.merge_all()
553    }

555    fn run_on_tables(
556        &mut self,
557        table_ids: &[TableId],
558        run: impl for<'a> Fn(TableId, &mut TableInfo, &DbView<'a>) -> bool + Sync,
559    ) {
560        if parallelize_db_level_op(self.total_size_estimate) {
561            let mut tables = Vec::with_capacity(table_ids.len());
562            for id in table_ids {
563                tables.push((*id, self.tables.take(*id).unwrap()));
564            }
565            let view = self.read_only_view();
566            parallel::for_each_mut(&mut tables, |_, (id, info)| {
567                if run(*id, info, &view) {
568                    self.notification_list.notify(*id);
569                }
570            });
571            for (id, info) in tables {
572                self.tables.insert(id, info);
573            }
574        } else {
575            for id in table_ids {
576                let mut info = self.tables.take(*id).unwrap();
577                let changed = {
578                    let view = self.read_only_view();
579                    run(*id, &mut info, &view)
580                };
581                if changed {
582                    self.notification_list.notify(*id);
583                }
584                self.tables.insert(*id, info);
585            }
586        }
587    }

634    /// Estimate the size of the table. If a constraint is provided, return an
635    /// estimate of the size of the subset of the table matching the constraint.
636    pub fn estimate_size(&self, table: TableId, c: Option<Constraint>) -> usize {
637        let table_info = self
638            .tables
639            .get(table)
640            .expect("table must be declared in the current database");
641        let table = &table_info.table;
642        if let Some(c) = c {
643            if let Some(sub) = table.fast_subset(&c) {
644                // In the case where a the constraint can be computed quickly,
645                // we do not filter for staleness, which may over-approximate.
646                sub.size()
647            } else {
648                table.refine_one(table.refine_live(table.all()), &c).size()
649            }
650        } else {
651            table.len()
652        }
653    }

689    /// A helper for merging all pending updates. Used to write to the database after updates have
690    /// been staged. Returns true if any tuples were added.
691    ///
692    /// Exposed for testing purposes.
693    ///
694    /// Useful for out-of-band insertions into the database.
695    pub fn merge_all(&mut self) -> bool {
696        let mut ever_changed = false;
697        let do_parallel = parallelize_db_level_op(self.total_size_estimate);
698        let mut to_merge = IndexSet::default();
699        // Tables modified during this `merge_all` call. Only these need their cached indexes reset
700        // at the end so future reads refresh them.
701        let mut touched: IndexSet<TableId> = IndexSet::default();
702        loop {
703            to_merge.clear();
704            let to_merge_vec = self.notification_list.reset();
705            touched.extend(to_merge_vec.iter().copied());
706            if to_merge_vec.len() < 4 {
707                ever_changed |= self.merge_simple(to_merge_vec, &mut touched);
708                break;
709            }
710            for table in to_merge_vec {
711                to_merge.insert(table);
712            }
713
714            let mut changed = false;
715            let mut tables_merging = DenseIdMap::<
716                TableId,
717                (
718                    // The info needed to merge this table.
719                    Option<TableInfo>,
720                    // Pre-allocated write buffers, according to the tables declared write
721                    // dependencies.
722                    DenseIdMap<TableId, Box<dyn MutationBuffer>>,
723                ),
724            >::with_capacity(self.tables.n_ids());
725            for stratum in self.deps.strata() {
726                // Initialize the write dependencies first.
727                for table in stratum.intersection(&to_merge).copied() {
728                    let mut bufs = DenseIdMap::default();
729                    for dep in self.deps.write_deps(table) {
730                        if let Some(info) = self.tables.get(dep) {
731                            bufs.insert(dep, info.table.new_buffer());
732                        }
733                    }
734                    tables_merging.insert(table, (None, bufs));
735                }
736                // Then initialize read dependencies (this two-phase structure is why we have an
737                // Option in the tables_merging map).
738                for table in stratum.intersection(&to_merge).copied() {
739                    let val = self.tables.unwrap_val(table);
740                    // Maintain `total_size_estimate` incrementally (subtract now, add
741                    // the post-merge length on drain below) so the reset loop no
742                    // longer re-sums every table.
743                    self.total_size_estimate =
744                        self.total_size_estimate.wrapping_sub(val.table.len());
745                    tables_merging[table].0 = Some(val);
746                }
747                let db = self.read_only_view();
748                changed |= if do_parallel {
749                    parallel::map_dense_id_map_mut(&mut tables_merging, |_, (info, buffers)| {
750                        let mut es = ExecutionState::new(db, mem::take(buffers));
751                        info.as_mut().unwrap().table.merge(&mut es).added || es.changed
752                    })
753                    .into_iter()
754                    .any(|changed| changed)
755                } else {
756                    tables_merging
757                        .iter_mut()
758                        .map(|(_, (info, buffers))| {
759                            let mut es = ExecutionState::new(db, mem::take(buffers));
760                            info.as_mut().unwrap().table.merge(&mut es).added || es.changed
761                        })
762                        .max()
763                        .unwrap_or(false)
764                };
765                for (id, (table, _)) in tables_merging.drain() {
766                    let val = table.unwrap();
767                    self.total_size_estimate =
768                        self.total_size_estimate.wrapping_add(val.table.len());
769                    self.tables.insert(id, val);
770                }
771            }
772            ever_changed |= changed;
773        }
774        // Reset the cached indexes of only the tables modified during this call so
775        // they refresh on next access; unmodified tables keep their still-valid
776        // cached indexes. `touched` must contain *every* table whose version bumped
777        // this call: `ResettableOnceLock::get_or_update` runs the index `refresh`
778        // only after a `reset()`, so a modified-but-unreset table would keep serving
779        // a stale cached index. It does — every merged table comes from
780        // `notification_list.reset()`, which is exactly what `touched` accumulates.
781        // `total_size_estimate` was maintained incrementally at each merge (above and
782        // in `merge_simple`), so we no longer re-sum every table here.
783        for table in touched.iter().copied() {
784            if let Some(info) = self.tables.get_mut(table) {
785                info.column_indexes.update(|_, ti| {
786                    Arc::get_mut(ti).unwrap().reset();
787                });
788                info.indexes.update(|_, ti| {
789                    Arc::get_mut(ti).unwrap().reset();
790                });
791            }
792        }
793        ever_changed
794    }

796    /// A "fast path" merge method that is not optimized for parallelism and does not respect read
797    /// and write dependencies. This ends up being faster than the full "strata-aware" option in
798    /// the body of `merge_all`.
799    fn merge_simple(
800        &mut self,
801        mut to_merge: SmallVec<[TableId; 4]>,
802        touched: &mut IndexSet<TableId>,
803    ) -> bool {
804        let mut changed = false;
805        while !to_merge.is_empty() {
806            for table_id in to_merge.iter().copied() {
807                let mut info = self.tables.unwrap_val(table_id);
808                // Maintain `total_size_estimate` incrementally (see `merge_all`'s
809                // reset loop, which no longer re-sums every table).
810                self.total_size_estimate = self.total_size_estimate.wrapping_sub(info.table.len());
811                let mut es = ExecutionState::new(self.read_only_view(), Default::default());
812                changed |= info.table.merge(&mut es).added || es.changed;
813                self.total_size_estimate = self.total_size_estimate.wrapping_add(info.table.len());
814                self.tables.insert(table_id, info);
815            }
816            to_merge = self.notification_list.reset();
817            touched.extend(to_merge.iter().copied());
818        }
819        changed
820    }

822    /// A low-level helper for merging pending updates to a particular function.
823    ///
824    /// Callers should prefer `merge_all`, as the process of merging the data
825    /// for a particular table may cause other updates to be buffered
826    /// elesewhere. The `merge_all` method runs merges to a fixed point to avoid
827    /// surprises here.
828    pub fn merge_table(&mut self, table: TableId) -> bool {
829        let mut info = self.tables.unwrap_val(table);
830        self.total_size_estimate = self.total_size_estimate.wrapping_sub(info.table.len());
831        let table_changed = info.table.merge(&mut ExecutionState::new(
832            self.read_only_view(),
833            Default::default(),
834        ));
835        self.total_size_estimate = self.total_size_estimate.wrapping_add(info.table.len());
836        self.tables.insert(table, info);
837        table_changed.added
838    }

935    pub(crate) fn process_constraints(
936        &self,
937        table: TableId,
938        cs: &[Constraint],
939    ) -> ProcessedConstraints {
940        let table_info = &self.tables[table];
941        let (mut subset, mut fast, mut slow) = table_info.table.split_fast_slow(cs);
942        slow.retain(|c| {
943            let (col, val) = match c {
944                Constraint::EqConst { col, val } => (*col, *val),
945                Constraint::Eq { .. }
946                | Constraint::LtConst { .. }
947                | Constraint::GtConst { .. }
948                | Constraint::LeConst { .. }
949                | Constraint::GeConst { .. } => return true,
950            };
951            // We are looking up by a constant: this is something we can build
952            // an index for as long as the column is cacheable.
953            if *table_info
954                .spec
955                .uncacheable_columns
956                .get(col)
957                .unwrap_or(&false)
958            {
959                return true;
960            }
961            // We have or will build an index: upgrade this constraint to
962            // 'fast'.
963            fast.push(c.clone());
964            let index = get_column_index_from_tableinfo(table_info, col);
965            match index.get().unwrap().get_subset(&val) {
966                Some(s) => {
967                    with_pool_set(|ps| subset.intersect(s, &ps.get_pool()));
968                }
969                None => {
970                    // There are no rows matching this key! We can constrain this to nothing.
971                    subset = Subset::empty();
972                }
973            }
974            // Remove this constraint from the slow list.
975            false
976        });
977        ProcessedConstraints { subset, fast, slow }
978    }

994    /// Remove every row from the given table.
995    ///
996    /// This is intended as a faster alternative to staging a per-row
997    /// `stage_remove` for every key in the table. The underlying [`Table::clear`]
998    /// implementation drops the row storage in bulk and bumps the table's major
999    /// generation, so any cached indexes/subsets observed by future readers will
1000    /// be lazily rebuilt against the now-empty table. Any pending staged
1001    /// inserts or removes for this table are dropped (they pre-dated the clear,
1002    /// so they no longer make sense once the table is empty).
1003    ///
1004    /// This method also resets the cached column- and key-indexes for the
1005    /// table so subsequent merges can take the `Arc::get_mut`-based reset path,
1006    /// matching the invariant maintained by [`Database::merge_all`].
1007    ///
1008    /// This does **not** flush pending changes for *other* tables; it is the
1009    /// caller's responsibility to call [`Database::merge_all`] beforehand if
1010    /// they need staged updates from a previous step to land before the clear.
1011    pub fn clear_table(&mut self, table: TableId) {
1012        let info = self
1013            .tables
1014            .get_mut(table)
1015            .expect("must access a table that has been declared in this database");
1016        let prev_len = info.table.len();
1017        info.table.clear();
1018        // The version bump from `clear` is enough on its own to make the
1019        // indexes self-refresh on next access (see `Index::refresh`). We still
1020        // reset them eagerly here so that the next `merge_all` sees the same
1021        // "indexes are resettable" state it expects after a successful merge.
1022        info.column_indexes.update(|_, ti| {
1023            if let Some(arc) = Arc::get_mut(ti) {
1024                arc.reset();
1025            }
1026        });
1027        info.indexes.update(|_, ti| {
1028            if let Some(arc) = Arc::get_mut(ti) {
1029                arc.reset();
1030            }
1031        });
1032        self.total_size_estimate = self.total_size_estimate.wrapping_sub(prev_len);
1033    }

1035    pub(crate) fn plan_query(&mut self, query: Query) -> Plan {
1036        let mut plan = plan::plan_query(query, ColumnCardEst::new(self));
1037        // Intern the successor family of every indexed access now, so nodes
1038        // shared across plans publish their children under ids that are fixed
1039        // before any rule set runs.
1040        let atoms = plan.atoms();
1041        plan.for_each_stages_mut(|stages| {
1042            let mut families = Vec::new();
1043            stages.for_each_indexed_access(|atom, columns, constraints| {
1044                let info = &mut self.tables[atoms[atom].table];
1045                let mut access = AccessFamilies::new();
1046                access.push(info.successor_family(columns[0], constraints));
1047                access.extend(
1048                    columns[1..]
1049                        .iter()
1050                        .map(|column| info.successor_family(*column, &[])),
1051                );
1052                families.push(access);
1053            });
1054            stages.set_families(families.into());
1055        });
1056        plan
1057    }

1068///
1069/// This is in a separate function to allow us to reuse it while already
1070/// borrowing a `TableInfo`.
1071fn get_index_from_tableinfo(table_info: &TableInfo, cols: &[ColumnId]) -> HashIndex {
1072    let index: Arc<_> = table_info.indexes.get_or_insert(cols.into(), || {
1073        Arc::new(ResettableOnceLock::new(Index::new(
1074            cols.to_vec(),
1075            TupleIndex::new(cols.len()),
1076        )))
1077    });
1078    index.get_or_update(|index| {
1079        index.refresh(table_info.table.as_ref());
1080    });
1081    debug_assert!(
1082        !index
1083            .get()
1084            .unwrap()
1085            .needs_refresh(table_info.table.as_ref())
1086    );
1087    index
1088}

1141/// A coarse cardinality estimate for a column of a table, used by the query
1142/// planner to decide which variable to eliminate next during tree
1143/// decomposition.
1144///
1145/// `table_size` is the number of rows in the (sub)table and `col_size` is the
1146/// number of distinct values in the column. Their ratio
1147/// (`table_size / col_size`) approximates the average number of rows that share
1148/// a given value of the column: a smaller ratio means the column is closer to
1149/// being unique and therefore cheaper to join on. [`ColUniqueness`] is ordered
1150/// by this ratio (see the [`Ord`] impl), so the planner prefers variables with
1151/// the most selective (most unique) columns.
1152#[derive(Copy, Clone, Debug)]
1153pub struct ColUniqueness {
1154    table_size: usize,
1155    col_size: usize,
1156}
1157
1158impl Default for ColUniqueness {
1159    fn default() -> ColUniqueness {
1160        ColUniqueness {
1161            table_size: 1,
1162            col_size: 1,
1163        }
1164    }
1165}
1166
1167impl ColUniqueness {
1168    #[allow(dead_code)] // not yet wired up into the planner
1169    fn scale(&self, subset_size: usize) -> ColUniqueness {
1170        if self.table_size == 0 || subset_size == 0 {
1171            return ColUniqueness {
1172                table_size: 0,
1173                col_size: 0,
1174            };
1175        }
1176        ColUniqueness {
1177            table_size: subset_size,
1178            col_size: self.col_size.saturating_mul(subset_size) / self.table_size,
1179        }
1180    }
1181    fn join(&self, other: &ColUniqueness) -> ColUniqueness {
1182        ColUniqueness {
1183            table_size: self.table_size.saturating_mul(other.table_size),
1184            col_size: self.col_size.max(other.col_size),
1185        }
1186    }
1187
1188    #[allow(dead_code)] // not yet wired up into the planner
1189    fn col_size(&self) -> usize {
1190        self.col_size
1191    }
1192}
